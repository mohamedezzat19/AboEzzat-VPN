name: Android Release Build

on:
  push:

  group: abo-vpn-release-${{ github.ref }}

jobs:

    name: Build signed Release APK
    runs-on: ubuntu-latest
    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Set up JDK 17
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: "17"

      - name: Set up Rust 1.90
        uses: actions-rust-lang/setup-rust-toolchain@v1
        with:
          toolchain: 1.90.0

        run: cargo test --manifest-path native/tun2proxy/Cargo.toml --lib --no-default-features

      - name: Check Rust engine for Android ABIs

          rustup target add aarch64-linux-android armv7-linux-androideabi x86_64-linux-android
          cargo check --manifest-path native/tun2proxy/Cargo.toml --lib --no-default-features --features udpgw --target aarch64-linux-android
          cargo check --manifest-path native/tun2proxy/Cargo.toml --lib --no-default-features --features udpgw --target armv7-linux-androideabi
          cargo check --manifest-path native/tun2proxy/Cargo.toml --lib --no-default-features --features udpgw --target x86_64-linux-android

      - name: Set up Android SDK
        uses: android-actions/setup-android@v3
        with:
          packages: "platform-tools platforms;android-35 build-tools;35.0.0 ndk;27.2.12479018 cmake;3.22.1"

      - name: Build tun2proxy for Android ABIs

          NDK_BIN="$ANDROID_HOME/ndk/27.2.12479018/toolchains/llvm/prebuilt/linux-x86_64/bin"

        uses: gradle/actions/setup-gradle@v4

          gradle-version: "8.9"

cmake_minimum_required(VERSION 3.22.1)
project(abz_protector LANGUAGES CXX)

add_library(abz_protector SHARED
    native_socket_protector.cpp
)

target_compile_features(abz_protector PRIVATE cxx_std_17)
target_compile_options(abz_protector PRIVATE
    -Wall
    -Wextra
    -Werror
)

# The Rust static library is built for the selected Android ABI by CI before
# Gradle invokes CMake. Refuse to produce a JNI-only APK if it is missing.
if(ANDROID_ABI STREQUAL "arm64-v8a")
    set(TUN2PROXY_RUST_TARGET "aarch64-linux-android")
elseif(ANDROID_ABI STREQUAL "armeabi-v7a")
    set(TUN2PROXY_RUST_TARGET "armv7-linux-androideabi")
elseif(ANDROID_ABI STREQUAL "x86_64")
    set(TUN2PROXY_RUST_TARGET "x86_64-linux-android")
else()
    message(FATAL_ERROR "Unsupported Android ABI for tun2proxy: ${ANDROID_ABI}")
endif()

set(TUN2PROXY_STATIC_LIB
    "${CMAKE_CURRENT_LIST_DIR}/../../../../native/tun2proxy/target/${TUN2PROXY_RUST_TARGET}/release/libtun2proxy.a"
)
if(NOT EXISTS "${TUN2PROXY_STATIC_LIB}")
    message(FATAL_ERROR "Missing patched tun2proxy static library: ${TUN2PROXY_STATIC_LIB}")
endif()

add_library(tun2proxy_rust STATIC IMPORTED GLOBAL)
set_target_properties(tun2proxy_rust PROPERTIES
    IMPORTED_LOCATION "${TUN2PROXY_STATIC_LIB}"
)

target_link_libraries(abz_protector PRIVATE
    tun2proxy_rust
    log
    android
    dl
    m
)

        PROTECT_LOGE("protect callback unavailable: fd=%d service=%p method=%p", fd, service, method);
        return false;
    }
    const jboolean allowed = env->CallBooleanMethod(service, method, static_cast<jint>(fd));
    if (env->ExceptionCheck()) {
        PROTECT_LOGE("Java protectNativeSocket threw for fd=%d", fd);
        env->ExceptionDescribe();
        env->ExceptionClear();
        return false;
    }
    if (allowed != JNI_TRUE) {
        PROTECT_LOGE("VpnService.protect returned false for fd=%d", fd);
        return false;
    }
    return true;
}

void on_native_traffic_status(const Tun2ProxyTrafficStatus* status, void*) {
    if (status == nullptr || g_vm == nullptr) return;

    jobject service = nullptr;
    jmethodID method = nullptr;
    {
        std::lock_guard<std::mutex> lock(g_mutex);
        if (g_shutting_down || g_service == nullptr || g_traffic_method == nullptr) return;
        ++g_in_flight;
        service = g_service;
        method = g_traffic_method;
    }

    JNIEnv* env = nullptr;
    bool attached_here = false;
    const jint get_env = g_vm->GetEnv(reinterpret_cast<void**>(&env), JNI_VERSION_1_6);
    if (get_env == JNI_EDETACHED) {
        if (g_vm->AttachCurrentThread(&env, nullptr) == JNI_OK) {
            attached_here = true;
        } else {
            PROTECT_LOGE("AttachCurrentThread failed in traffic callback");
        }
    } else if (get_env != JNI_OK) {
        PROTECT_LOGE("GetEnv failed (%d) in traffic callback", get_env);
        env = nullptr;
    }

    if (env != nullptr) {
        env->CallVoidMethod(service, method, static_cast<jlong>(status->tx), static_cast<jlong>(status->rx));
        if (env->ExceptionCheck()) env->ExceptionClear();
    }
    if (attached_here) g_vm->DetachCurrentThread();

    {
        std::lock_guard<std::mutex> lock(g_mutex);
        if (g_in_flight > 0) --g_in_flight;
        if (g_in_flight == 0) g_idle.notify_all();
    }
}

bool protect_fd_from_native(int fd) {
    if (fd < 0 || g_vm == nullptr) { PROTECT_LOGE("invalid fd or JavaVM: fd=%d vm=%p", fd, g_vm); return false; }

    jobject service = nullptr;
    jmethodID method = nullptr;
    {
        std::lock_guard<std::mutex> lock(g_mutex);
        if (g_shutting_down || g_service == nullptr || g_protect_method == nullptr) {
            PROTECT_LOGE("service callback not bound: fd=%d shutting_down=%d service=%p method=%p", fd, g_shutting_down, g_service, g_protect_method);
            return false;
        }
        ++g_in_flight;
        service = g_service;
        method = g_protect_method;
    }

    JNIEnv* env = nullptr;
    bool attached_here = false;
    bool allowed = false;
    const jint get_env = g_vm->GetEnv(reinterpret_cast<void**>(&env), JNI_VERSION_1_6);
    if (get_env == JNI_EDETACHED) {
        if (g_vm->AttachCurrentThread(&env, nullptr) == JNI_OK) {
            attached_here = true;
        } else {
            PROTECT_LOGE("AttachCurrentThread failed for fd=%d", fd);
        }
    } else if (get_env != JNI_OK) {
        PROTECT_LOGE("GetEnv failed (%d) for fd=%d", get_env, fd);
        env = nullptr;
    }

    if (env != nullptr) {
        allowed = protect_fd(env, service, method, fd);
    } else {
        PROTECT_LOGE("JNIEnv unavailable for fd=%d", fd);
    }

    if (attached_here) {
        g_vm->DetachCurrentThread();
    }

    {
        std::lock_guard<std::mutex> lock(g_mutex);
        if (g_in_flight > 0) --g_in_flight;
        if (g_in_flight == 0) g_idle.notify_all();
    }
    return allowed;
}

extern "C" JNIEXPORT jboolean JNICALL
Java_com_aboezzat_vpn_vpn_NativeSocketProtectorBridge_nativeBindService(

    if (env == nullptr || service == nullptr) return JNI_FALSE;

    if (service_class == nullptr) return JNI_FALSE;
    jmethodID method = env->GetMethodID(service_class, "protectNativeSocket", "(I)Z");
    jmethodID traffic_method = env->GetMethodID(service_class, "onNativeTrafficStatus", "(JJ)V");
    env->DeleteLocalRef(service_class);
    if (method == nullptr || traffic_method == nullptr || env->ExceptionCheck()) {
        if (env->ExceptionCheck()) env->ExceptionClear();
        return JNI_FALSE;
    }

    jobject global_service = env->NewGlobalRef(service);
    if (global_service == nullptr) return JNI_FALSE;

    std::lock_guard<std::mutex> lock(g_mutex);
    if (g_service != nullptr || g_shutting_down || g_in_flight != 0) {
        env->DeleteGlobalRef(global_service);
        return JNI_FALSE;
    }
    g_service = global_service;
    g_protect_method = method;
    g_traffic_method = traffic_method;
    return JNI_TRUE;
}

extern "C" JNIEXPORT jint JNICALL
Java_com_aboezzat_vpn_vpn_NativeSocketProtectorBridge_nativeRunEngine(
        JNIEnv* env, jclass, jstring cli_args, jint tun_mtu, jboolean packet_information) {
    if (env == nullptr || cli_args == nullptr || tun_mtu < 576 || tun_mtu > 65535) {
        return -10;
    }
    const char* chars = env->GetStringUTFChars(cli_args, nullptr);
    if (chars == nullptr) return -11;

    {
        std::lock_guard<std::mutex> lock(g_engine_mutex);
        if (g_engine_running) {
            env->ReleaseStringUTFChars(cli_args, chars);
            return -12;
        }
        if (!g_engine_pending_start) {
            env->ReleaseStringUTFChars(cli_args, chars);
            return -14;
        }
        if (g_engine_stop_requested) {
            g_engine_pending_start = false;
            g_engine_stop_requested = false;
            env->ReleaseStringUTFChars(cli_args, chars);
            return -13;
        }
        g_engine_pending_start = false;
        g_engine_running = true;
    }

    tun2proxy_set_traffic_status_callback(1, &on_native_traffic_status, nullptr);
    const int result = tun2proxy_run_with_cli_args(
        chars,
        static_cast<unsigned short>(tun_mtu),
        packet_information == JNI_TRUE
    );
    env->ReleaseStringUTFChars(cli_args, chars);
    // Mark the native run idle before clearing a stop request that raced
    // with Rust dropping its cancellation token. Stop calls serialize on the
    // same mutex, so no late request can survive into the next run.
    {
        std::lock_guard<std::mutex> lock(g_engine_mutex);
        g_engine_running = false;
        g_engine_pending_start = false;
        g_engine_stop_requested = false;
    }
    tun2proxy_clear_stop_request();
    return result;
}

extern "C" JNIEXPORT jint JNICALL
Java_com_aboezzat_vpn_vpn_NativeSocketProtectorBridge_nativeStopEngine(
        JNIEnv*, jclass) {
    std::lock_guard<std::mutex> lock(g_engine_mutex);
    if (!g_engine_running) {
        // Only queue a stop while Kotlin has armed a specific pending run.
        if (g_engine_pending_start) {
            g_engine_stop_requested = true;
            return 0;
        }
        return -1;
    }
    // Serialize the Rust cancellation request with the transition to IDLE.
    return tun2proxy_stop();
}

}

package com.aboezzat.vpn.core

/**
 * Resolves a network connection's Android UID to the route policy selected
 * for the package(s) that own that UID.
 *
 * Android routes VPN traffic by UID, not by package. Multiple packages can
 * share one UID, so conflicting explicit policies cannot be enforced
 * independently and are reported as a conflict instead of choosing one
 * package arbitrarily.
 */
fun interface UidPackageLookup {
    fun packagesForUid(uid: Int): List<String>
}

enum class AppRouteResolutionReason {
    EXPLICIT_POLICY,
    DEFAULT_POLICY,
    UNKNOWN_UID,
    CONFLICTING_SHARED_UID_POLICIES
}

data class AppRouteResolution(
    val uid: Int,
    val packageNames: List<String>,
    val mode: RouteMode,
    val reason: AppRouteResolutionReason
)

class AppRoutePolicyResolver(
    private val packageLookup: UidPackageLookup,
    private val defaultMode: RouteMode = RouteMode.VPN_ONLY
) {
    fun resolve(uid: Int, routes: Map<String, RouteMode>): AppRouteResolution {
        if (uid <= 0) {
            return AppRouteResolution(
                uid = uid,
                packageNames = emptyList(),
                mode = defaultMode,
                reason = AppRouteResolutionReason.UNKNOWN_UID
            )
        }

        val packages = packageLookup.packagesForUid(uid)
            .asSequence()
            .filter { it.isNotBlank() }
            .distinct()
            .sorted()
            .toList()

        if (packages.isEmpty()) {
            return AppRouteResolution(
                uid = uid,
                packageNames = emptyList(),
                mode = defaultMode,
                reason = AppRouteResolutionReason.UNKNOWN_UID
            )
        }

        val explicitModes = packages.mapNotNull(routes::get).distinct()
        return when (explicitModes.size) {
            0 -> AppRouteResolution(
                uid, packages, defaultMode, AppRouteResolutionReason.DEFAULT_POLICY
            )
            1 -> AppRouteResolution(
                uid, packages, explicitModes.single(), AppRouteResolutionReason.EXPLICIT_POLICY
            )
            else -> AppRouteResolution(
                uid, packages, defaultMode,
                AppRouteResolutionReason.CONFLICTING_SHARED_UID_POLICIES
            )
        }
    }
}

import android.content.Context

import android.security.keystore.KeyProperties

import com.aboezzat.vpn.proxy.ProxyProtocol

import javax.crypto.spec.GCMParameterSpec

class ProxyProfileStore(context: Context) {
    private val prefs = context.getSharedPreferences("proxy_profiles_v1", Context.MODE_PRIVATE)
    @Volatile private var credentialReadFailure = false

    fun hasCredentialReadFailure(): Boolean = credentialReadFailure

    fun loadProfiles(): List<ProxyProfile> {
        return try {
            val array = JSONArray(prefs.getString("profiles", "[]"))
            buildList {
                for (i in 0 until array.length()) {
                    val item = array.getJSONObject(i)
                    val protocol = runCatching {
                        ProxyProtocol.valueOf(item.optString("protocol", "HTTP_CONNECT"))
                    }.getOrDefault(ProxyProtocol.HTTP_CONNECT)
                    add(
                        ProxyProfile(
                            name = item.optString("name"),
                            host = item.optString("host"),
                            port = item.optInt("port"),
                            protocol = protocol,
                            username = decrypt(item.optString("username", "")),
                            password = decrypt(item.optString("password", "")),
                            remoteDns = item.optBoolean("remoteDns", true)
                        )
                    )
                }
            }
        } catch (_: Exception) {
            credentialReadFailure = true
            emptyList()
        }
    }

    /**
     * Refuses to overwrite stored profiles if a credential could not be
     * decrypted, preserving the original encrypted values on disk.
     */
    fun saveProfiles(profiles: List<ProxyProfile>): Boolean {
        if (credentialReadFailure) return false
        return try {
            val array = JSONArray()
            profiles.forEach { profile ->
                array.put(
                    JSONObject().apply {
                        put("name", profile.name)
                        put("host", profile.host)
                        put("port", profile.port)
                        put("protocol", profile.protocol.name)
                        put("username", encrypt(profile.username.orEmpty()))
                        put("password", encrypt(profile.password.orEmpty()))
                        put("remoteDns", profile.remoteDns)
                    }
                )
            }
            prefs.edit().putString("profiles", array.toString()).commit()
        } catch (_: Exception) {
            false
        }
    }

    fun activeProfileName(): String = prefs.getString("active_profile", "") ?: ""

    fun setActiveProfile(name: String) {
        prefs.edit().putString("active_profile", name).apply()
    }

    private fun encrypt(value: String): String {
        if (value.isEmpty()) return ""
        val cipher = Cipher.getInstance("AES/GCM/NoPadding")
        cipher.init(Cipher.ENCRYPT_MODE, getOrCreateKey())
        val encrypted = cipher.doFinal(value.toByteArray(Charsets.UTF_8))
        return Base64.encodeToString(cipher.iv + encrypted, Base64.NO_WRAP)
    }

    private fun decrypt(value: String): String? {
        if (value.isEmpty()) return null
        return try {
            val bytes = Base64.decode(value, Base64.NO_WRAP)
            val iv = bytes.copyOfRange(0, 12)
            val encrypted = bytes.copyOfRange(12, bytes.size)
            val cipher = Cipher.getInstance("AES/GCM/NoPadding")
            cipher.init(Cipher.DECRYPT_MODE, getOrCreateKey(), GCMParameterSpec(128, iv))
            String(cipher.doFinal(encrypted), Charsets.UTF_8)
        } catch (_: Exception) {
            credentialReadFailure = true
            null
        }
    }

    private fun getOrCreateKey(): SecretKey {
        val keyStore = KeyStore.getInstance("AndroidKeyStore").apply { load(null) }
        (keyStore.getKey(KEY_ALIAS, null) as? SecretKey)?.let { return it }
        val generator = KeyGenerator.getInstance(KeyProperties.KEY_ALGORITHM_AES, "AndroidKeyStore")
        generator.init(
            KeyGenParameterSpec.Builder(
                KEY_ALIAS,
                KeyProperties.PURPOSE_ENCRYPT or KeyProperties.PURPOSE_DECRYPT
            )
                .setBlockModes(KeyProperties.BLOCK_MODE_GCM)
                .setEncryptionPaddings(KeyProperties.ENCRYPTION_PADDING_NONE)
                .setRandomizedEncryptionRequired(true)
                .build()
        )
        return generator.generateKey()
    }

    private companion object {
        const val KEY_ALIAS = "aboezzat_vpn_proxy_credentials"
    }
}

            updateError = error.message ?: "Could not check for updates."
        } finally {
            updateChecking = false

    fun checkForAppUpdate() {
        updateChecking = true
        updateError = ""
        updateScope.launch {
            try {
                updateInfo = withContext(Dispatchers.IO) {
                    GitHubReleaseUpdater.checkForUpdate(context)
                }
            } catch (error: Exception) {
                updateError = error.message ?: "Could not check for updates."
            } finally {
                updateChecking = false
            }
        }
    }

    fun downloadAppUpdate() {
        val info = updateInfo ?: return
        updateDownloading = true
        updateProgress = 0f
        updateDownloadedBytes = 0L
        updateTotalBytes = 0L
        updateError = ""
        downloadedApk = null
        updateScope.launch {
            try {
                downloadedApk = withContext(Dispatchers.IO) {
                    GitHubReleaseUpdater.download(context, info) { downloaded, total ->
                        mainHandler.post {
                            updateDownloadedBytes = downloaded
                            updateTotalBytes = total
                            updateProgress = if (total > 0L) (downloaded.toFloat() / total).coerceIn(0f, 1f) else 0f
                        }
                    }
                }
            } catch (error: Exception) {
                updateError = error.message ?: "APK download failed."
            } finally {
                updateDownloading = false
            }
        }
    }

    fun installAppUpdate(file: File) {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O && !context.packageManager.canRequestPackageInstalls()) {
            updateError = "Allow this app to install unknown apps, then tap Install update again."
            context.startActivity(
                Intent(Settings.ACTION_MANAGE_UNKNOWN_APP_SOURCES, Uri.parse("package:${context.packageName}"))
            )
            return
        }
        try {
            val uri = FileProvider.getUriForFile(context, "${context.packageName}.fileprovider", file)
            val installIntent = Intent(Intent.ACTION_VIEW).apply {
                setDataAndType(uri, "application/vnd.android.package-archive")
                addFlags(Intent.FLAG_ACTIVITY_NEW_TASK or Intent.FLAG_GRANT_READ_URI_PERMISSION)
            }
            context.startActivity(installIntent)
        } catch (error: Exception) {
            updateError = error.message ?: "Could not open Android package installer."
        }
    }

    LaunchedEffect(activeName) {
        proxyConnected = false
        proxyTesting = false
        proxyStatus = "Not tested"
    }
    val tunnelResultReceiver = remember(mainHandler) {
        object : ResultReceiver(mainHandler) {
            override fun onReceiveResult(resultCode: Int, resultData: Bundle?) {
                if (resultCode == AboEzzatVpnService.RESULT_TRAFFIC_UPDATE) {
                    tunnelTxBytes = resultData?.getLong(AboEzzatVpnService.EXTRA_TX_BYTES) ?: tunnelTxBytes
                    tunnelRxBytes = resultData?.getLong(AboEzzatVpnService.EXTRA_RX_BYTES) ?: tunnelRxBytes
                    return
                }
                tunnelStatus = resultData?.getString(AboEzzatVpnService.EXTRA_MESSAGE)
                    ?: "Tunnel status updated."
                when (resultCode) {
                    AboEzzatVpnService.RESULT_STARTED,
                    AboEzzatVpnService.RESULT_ALREADY_ACTIVE -> {
                        tunnelActive = true
                        tunnelConnecting = false
                    }
                    AboEzzatVpnService.RESULT_STOPPING,
                    AboEzzatVpnService.RESULT_STOP_FAILED -> {
                        tunnelActive = resultCode == AboEzzatVpnService.RESULT_STOP_FAILED
                        tunnelConnecting = false
                    }
                    AboEzzatVpnService.RESULT_STOPPED,
                    AboEzzatVpnService.RESULT_FAILED -> {
                        tunnelActive = false
                        tunnelConnecting = false
                    }
                }
                notice = tunnelStatus
            }
        }
    }

    fun sendTunnelCommand(action: String, profile: ProxyProfile? = null) {
        val serviceIntent = Intent(context, AboEzzatVpnService::class.java).apply {
            this.action = action
            putExtra(AboEzzatVpnService.EXTRA_RESULT_RECEIVER, tunnelResultReceiver)
            if (profile != null) {
                putExtra(AboEzzatVpnService.EXTRA_NAME, profile.name)
                putExtra(AboEzzatVpnService.EXTRA_HOST, profile.host)
                putExtra(AboEzzatVpnService.EXTRA_PORT, profile.port)
                putExtra(AboEzzatVpnService.EXTRA_PROTOCOL, profile.protocol.name)
                putExtra(AboEzzatVpnService.EXTRA_USERNAME, profile.username)
                putExtra(AboEzzatVpnService.EXTRA_PASSWORD, profile.password)
                putExtra(AboEzzatVpnService.EXTRA_REMOTE_DNS, profile.remoteDns)
            }
        }
        try {
            if (action == AboEzzatVpnService.ACTION_START_TUNNEL) {
                context.startForegroundService(serviceIntent)
            } else {
                context.startService(serviceIntent)
            }
        } catch (error: Exception) {
            if (action == AboEzzatVpnService.ACTION_START_TUNNEL) tunnelConnecting = false
            notice = error.message ?: "Could not send the tunnel command."
        }
    }

    fun testProxyProfile(profile: ProxyProfile) {
        proxyTesting = true
        proxyConnected = false
        proxyStatus = "Testing"
        val resultReceiver = object : ResultReceiver(mainHandler) {
            override fun onReceiveResult(resultCode: Int, resultData: Bundle?) {
                val success = resultData?.getBoolean(
                    AboEzzatVpnService.EXTRA_SUCCESS,
                    resultCode == AboEzzatVpnService.RESULT_OK
                ) ?: (resultCode == AboEzzatVpnService.RESULT_OK)
                val message = resultData?.getString(AboEzzatVpnService.EXTRA_MESSAGE)
                    ?: "Proxy service returned no diagnostic message."
                val latency = resultData?.getLong(AboEzzatVpnService.EXTRA_LATENCY) ?: 0L
                proxyConnected = success
                proxyTesting = false
                proxyStatus = if (success) "Connected" else "Disconnected"
                testResults[profile.name] = ProxyTestResult(success, message, latency)
                notice = if (success) "$message (${latency} ms)" else message
            }
        }
        val serviceIntent = Intent(context, AboEzzatVpnService::class.java).apply {
            action = AboEzzatVpnService.ACTION_TEST_UPSTREAM_PROXY
            putExtra(AboEzzatVpnService.EXTRA_RESULT_RECEIVER, resultReceiver)
            putExtra(AboEzzatVpnService.EXTRA_NAME, profile.name)
            putExtra(AboEzzatVpnService.EXTRA_HOST, profile.host)
            putExtra(AboEzzatVpnService.EXTRA_PORT, profile.port)
            putExtra(AboEzzatVpnService.EXTRA_PROTOCOL, profile.protocol.name)
            putExtra(AboEzzatVpnService.EXTRA_USERNAME, profile.username)
            putExtra(AboEzzatVpnService.EXTRA_PASSWORD, profile.password)
            putExtra(AboEzzatVpnService.EXTRA_REMOTE_DNS, profile.remoteDns)
        }
        try {
            context.startService(serviceIntent)
        } catch (error: Exception) {
            proxyTesting = false
            proxyConnected = false
            proxyStatus = "Disconnected"
            notice = error.message ?: "Could not start the VPN service for proxy testing."
        }
    }

    fun connectSelectedProfile(targetProfile: ProxyProfile? = null) {
        if (Build.VERSION.SDK_INT >= 33 &&
            ContextCompat.checkSelfPermission(context, Manifest.permission.POST_NOTIFICATIONS) != PackageManager.PERMISSION_GRANTED &&
            !notificationPermissionRequested
        ) {
            notificationPermissionLauncher.launch(Manifest.permission.POST_NOTIFICATIONS)
            return
        }
        val profile = targetProfile ?: profiles.firstOrNull { it.name == activeName }
        if (profile == null) {
            notice = "Add and select a proxy profile first."
            return
        }
        val prepareIntent = VpnService.prepare(context)
        if (prepareIntent != null) {
            vpnPermissionLauncher.launch(prepareIntent)
            notice = "Grant Android VPN permission, then tap Connect again."
            return
        }
        vpnPermissionGranted = true
        tunnelTxBytes = 0L
        tunnelRxBytes = 0L
        tunnelConnecting = true
        sendTunnelCommand(AboEzzatVpnService.ACTION_START_TUNNEL, profile)
    }

                    notice = "Profile saved on this device"
                }
            }
        )

@Composable
private fun tx(english: String): String {
    if (LocalConfiguration.current.locales[0].language != "ar") return english
    return when {
        english.startsWith("Allow this app to install unknown apps") -> "اسمح للتطبيق بتثبيت التطبيقات غير المعروفة، ثم اضغط تثبيت التحديث مرة أخرى."
        english.startsWith("Could not check for updates") -> "تعذر فحص التحديثات: " + english.substringAfter(":").trim()
        english.startsWith("APK download failed") -> "فشل تنزيل الحزمة: " + english.substringAfter(":").trim()
        english.startsWith("Active profile: ") -> "ملف البروكسي النشط: " + english.removePrefix("Active profile: ")
        english.startsWith("Route for ") -> "مسار التطبيق: " + english.removePrefix("Route for ")
        english.startsWith("Proxy service returned no diagnostic message.") -> "لم تُرجع خدمة البروكسي أي رسالة تشخيصية."
        english.startsWith("Could not start the VPN service for proxy testing.") -> "تعذر تشغيل خدمة VPN لاختبار البروكسي."
        english.startsWith("HTTP proxy rejected CONNECT:") -> "رفض بروكسي HTTP طلب CONNECT: " + english.substringAfter(":").trim()
        english.startsWith("SOCKS5 CONNECT failed") -> "فشل اتصال SOCKS5: " + english.substringAfter("failed")
        english.startsWith("HTTP/1.1 407") -> "رفض البروكسي المصادقة (HTTP 407)."
        english.startsWith("HTTP/1.0 407") -> "رفض البروكسي المصادقة (HTTP 407)."
        else -> arabicUi[english] ?: english
    }
}

private fun formatUpdateBytes(bytes: Long): String {
    if (bytes < 1024L) return "$bytes B"
    val mb = bytes / (1024.0 * 1024.0)
    return String.format(java.util.Locale.US, "%.1f MB", mb)
}

@Composable
private fun ConnectionCircle(
    title: String,
    connected: Boolean?,
    connecting: Boolean,
    onClick: () -> Unit
) {
    val stateColor = when {
        connecting -> Color(0xFF2867A8)
        connected == true -> Color(0xFF168B55)
        connected == false -> Color(0xFF303846)
        else -> Color(0xFF303846)
    }
    val statusText = when {
        connecting -> "Connecting…"
        connected == true -> "Running"
        else -> "Disconnected"
    }
    Column(
        horizontalAlignment = Alignment.CenterHorizontally,
        verticalArrangement = Arrangement.spacedBy(7.dp)
    ) {
        Surface(
            modifier = Modifier.size(100.dp).clickable(enabled = !connecting, onClick = onClick),
            shape = CircleShape,
            color = stateColor,
            border = if (connected != true && !connecting) BorderStroke(1.dp, Color(0xFF718096)) else null
        ) {
            Box(contentAlignment = Alignment.Center) {
                if (connecting) {
                    CircularProgressIndicator(
                        modifier = Modifier.size(31.dp),
                        color = Color.White,
                        strokeWidth = 3.dp
                    )
                } else {
                    Text(
                        tx(statusText),
                        color = Color.White,
                        fontSize = 13.sp,
                        fontWeight = FontWeight.Bold,
                        textAlign = androidx.compose.ui.text.style.TextAlign.Center,
                        maxLines = 1
                    )
                }
            }
        }
        Text(tx(title), color = Color.White, fontSize = 14.sp, fontWeight = FontWeight.Bold)
    }
}

                icon = info.loadIcon(context.packageManager)

import java.io.BufferedInputStream
import java.io.BufferedOutputStream

data class ProxyTestResult(
    val success: Boolean,
    val message: String,
    val latencyMs: Long
)

object ProxyConnectionTester {
    private const val CONNECT_TIMEOUT_MS = 7000
    private const val READ_TIMEOUT_MS = 7000
    private const val TEST_HOST = "example.com"
    private const val TEST_PORT = 443

    fun test(profile: ProxyProfile, protectSocket: ((Socket) -> Boolean)? = null): ProxyTestResult {
        profile.validate()?.let { return ProxyTestResult(false, it, 0) }
        val start = System.nanoTime()
        fun elapsedMs() = (System.nanoTime() - start) / 1_000_000
        return try {
            Socket().use { socket ->
                // Force Android to create the underlying file descriptor before
                // VpnService.protect(Socket) is called. Keep protection before connect.
                socket.bind(InetSocketAddress(0))
                if (protectSocket != null && !protectSocket(socket)) {
                    throw IOException("Could not protect the upstream socket from VPN routing.")
                }
                socket.connect(InetSocketAddress(profile.host, profile.port), CONNECT_TIMEOUT_MS)
                socket.soTimeout = READ_TIMEOUT_MS
                val input = BufferedInputStream(socket.getInputStream())
                val output = BufferedOutputStream(socket.getOutputStream())
                when (profile.protocol) {
                    ProxyProtocol.HTTP_CONNECT -> testHttpConnect(profile, input, output)
                    ProxyProtocol.SOCKS5 -> testSocks5(profile, input, output)
                }

                // A successful proxy handshake is not enough: verify destination TLS
                // and read an actual HTTP response through the established tunnel.
                val ssl = (SSLSocketFactory.getDefault() as SSLSocketFactory)
                    .createSocket(socket, TEST_HOST, TEST_PORT, false) as SSLSocket
                ssl.use { tls ->
                    tls.soTimeout = READ_TIMEOUT_MS
                    tls.sslParameters = tls.sslParameters.apply {
                        endpointIdentificationAlgorithm = "HTTPS"
                    }
                    tls.startHandshake()
                    val request = "GET / HTTP/1.1\r\n" +
                        "Host: $TEST_HOST\r\n" +
                        "User-Agent: AboEzzat-VPN-Diagnostic/1.0\r\n" +
                        "Connection: close\r\n\r\n"
                    tls.outputStream.write(request.toByteArray(StandardCharsets.US_ASCII))
                    tls.outputStream.flush()
                    val response = BufferedInputStream(tls.inputStream)
                    val headers = readHttpHeaders(response)
                    val status = headers.lineSequence().firstOrNull().orEmpty()
                    if (!status.contains(" 200 ")) {
                        throw IOException("TLS connected, but destination returned: $status")
                    }
                    if (response.read() < 0) {
                        throw IOException("TLS connected, but no response body data was received.")
                    }
                }
            }
            ProxyTestResult(
                true,
                "Proxy CONNECT, destination TLS and HTTPS data transfer succeeded.",
                elapsedMs()
            )
        } catch (e: javax.net.ssl.SSLException) {
            ProxyTestResult(false, "Proxy CONNECT succeeded or started, but destination TLS failed: ${e.message?.take(100) ?: "TLS error"}", elapsedMs())
        } catch (e: java.net.SocketTimeoutException) {
            ProxyTestResult(false, "Timed out during proxy connection, TLS or HTTPS data transfer.", elapsedMs())
        } catch (e: java.net.UnknownHostException) {
            ProxyTestResult(false, "Could not resolve the proxy server or destination address.", elapsedMs())
        } catch (e: Exception) {
            ProxyTestResult(false, e.message?.take(140) ?: "Proxy HTTPS test failed.", elapsedMs())
        }
    }

    private fun testHttpConnect(
        profile: ProxyProfile,
        input: BufferedInputStream,
        output: BufferedOutputStream
    ) {
        val auth = if (!profile.username.isNullOrEmpty()) {
            val raw = "${profile.username}:${profile.password.orEmpty()}"
            val encoded = Base64.getEncoder().encodeToString(raw.toByteArray(StandardCharsets.UTF_8))
            "Proxy-Authorization: Basic $encoded\r\n"
        } else ""
        val request = "CONNECT $TEST_HOST:$TEST_PORT HTTP/1.1\r\n" +
            "Host: $TEST_HOST:$TEST_PORT\r\n" +
            auth +
            "Proxy-Connection: Keep-Alive\r\n\r\n"
        output.write(request.toByteArray(StandardCharsets.US_ASCII))
        output.flush()
        val response = readHttpHeaders(input)
        val status = response.lineSequence().firstOrNull().orEmpty()
        when {
            status.contains(" 200 ") -> Unit
            status.contains(" 407 ") -> throw SecurityException("Proxy authentication required or credentials rejected (HTTP 407).")
            else -> throw IllegalStateException("HTTP proxy rejected CONNECT: $status")
        }
    }

    private fun testSocks5(
        profile: ProxyProfile,
        input: BufferedInputStream,
        output: BufferedOutputStream
    ) {
        val hasCredentials = !profile.username.isNullOrEmpty()
        output.write(if (hasCredentials) byteArrayOf(0x05, 0x02, 0x00, 0x02) else byteArrayOf(0x05, 0x01, 0x00))
        output.flush()
        val version = input.read()
        val method = input.read()
        if (version != 5) throw IllegalStateException("Server is not a SOCKS5 proxy.")
        when (method) {
            0 -> Unit
            2 -> {
                if (!hasCredentials) throw SecurityException("SOCKS5 server requires a username and password.")
                val user = profile.username!!.toByteArray(StandardCharsets.UTF_8)
                val pass = profile.password.orEmpty().toByteArray(StandardCharsets.UTF_8)
                require(user.size in 1..255 && pass.size in 1..255) { "SOCKS5 credentials must be 1–255 bytes." }
                output.write(byteArrayOf(0x01, user.size.toByte()))
                output.write(user)
                output.write(pass.size)
                output.write(pass)
                output.flush()
                val authVersion = input.read()
                val authStatus = input.read()
                if (authVersion != 1 || authStatus != 0) {
                    throw SecurityException("SOCKS5 authentication failed.")
                }
            }
            0xFF -> throw SecurityException("SOCKS5 server rejected all authentication methods.")
            else -> throw IllegalStateException("Unsupported SOCKS5 authentication method.")
        }

        val address = if (profile.remoteDns) {
            val host = TEST_HOST.toByteArray(StandardCharsets.US_ASCII)
            byteArrayOf(0x03, host.size.toByte()) + host
        } else {
            val ip = InetAddress.getByName(TEST_HOST).address
            if (ip.size == 4) byteArrayOf(0x01) + ip else byteArrayOf(0x04) + ip
        }
        output.write(byteArrayOf(0x05, 0x01, 0x00) + address + byteArrayOf((TEST_PORT shr 8).toByte(), TEST_PORT.toByte()))
        output.flush()

        val responseVersion = input.read()
        val reply = input.read()
        input.read()
        val addressType = input.read()
        if (responseVersion != 5) throw IllegalStateException("Invalid SOCKS5 response.")
        if (reply != 0) throw IllegalStateException("SOCKS5 CONNECT failed (reply code $reply).")
        val addressLength = when (addressType) {
            1 -> 4
            4 -> 16
            3 -> input.read().also { if (it < 0) throw IllegalStateException("Incomplete SOCKS5 response.") }
            else -> throw IllegalStateException("Unknown SOCKS5 address type.")
        }
        repeat(addressLength + 2) {
            if (input.read() < 0) throw IllegalStateException("Incomplete SOCKS5 response.")
        }
    }

    private fun readHttpHeaders(input: BufferedInputStream): String {
        val bytes = ArrayList<Byte>()
        var matched = 0
        val terminator = byteArrayOf(13, 10, 13, 10)
        while (bytes.size < 16_384) {
            val next = input.read()
            if (next < 0) break
            bytes.add(next.toByte())
            if (next.toByte() == terminator[matched]) {
                matched++
                if (matched == terminator.size) break
            } else {
                matched = if (next == 13) 1 else 0
            }
        }
        if (matched != terminator.size) throw IllegalStateException("Incomplete HTTP proxy response.")
        return bytes.toByteArray().toString(StandardCharsets.US_ASCII)
    }
}

package com.aboezzat.vpn.proxy

import java.io.IOException
import java.net.Socket
import java.util.concurrent.ExecutorService

/**
 * Connects a client TCP socket to a destination through the configured
 * upstream proxy, then relays bytes in both directions.
 *
 * Stream-level bridge only: it does not create/read a TUN interface, recover
 * packet destinations, or enforce Android per-app routing.
 *
 * The returned session owns both sockets. The supplied manager and executor
 * remain owned by the caller.
 */
class ProxyTcpRelay(
    private val tunnelManager: ProxyTcpTunnelManager,
    private val relayExecutor: ExecutorService
) {
    @Throws(IOException::class)
    fun open(
        clientSocket: Socket,
        profile: ProxyProfile,
        destinationHost: String,
        destinationPort: Int,
        protectSocket: ((Socket) -> Boolean)? = null
    ): TcpStreamRelay.Session {
        val upstream = try {
            tunnelManager.openTunnel(
                profile = profile,
                destinationHost = destinationHost,
                destinationPort = destinationPort,
                protectSocket = protectSocket
            )
        } catch (error: Exception) {
            closeQuietly(clientSocket)
            throw error
        }

        return try {
            TcpStreamRelay.start(
                client = clientSocket,
                upstream = upstream,
                executor = relayExecutor,
                onClosed = { tunnelManager.release(upstream) }
            )
        } catch (error: Exception) {
            tunnelManager.release(upstream)
            closeQuietly(clientSocket)
            throw error
        }
    }

    private fun closeQuietly(socket: Socket) {
        try {
            socket.close()
        } catch (_: IOException) {
            // Preserve the original open/relay failure.
        }
    }
}

package com.aboezzat.vpn.proxy

import java.io.IOException
import java.io.InputStream
import java.io.OutputStream
import java.net.InetAddress
import java.net.InetSocketAddress
import java.net.Socket
import java.nio.charset.StandardCharsets
import java.util.Base64

/**
 * Opens a TCP tunnel through an upstream HTTP CONNECT or SOCKS5 proxy.
 *
 * The returned socket belongs to the caller and must be closed by the caller.
 * This class does not create a VPN/TUN interface and does not forward packets.
 */
object ProxyTcpTunnel {
    private const val DEFAULT_CONNECT_TIMEOUT_MS = 7_000
    private const val DEFAULT_READ_TIMEOUT_MS = 7_000
    private const val MAX_HTTP_HEADER_BYTES = 16_384
    private const val DESTINATION_PORT_MAX = 65_535

    @Throws(IOException::class)
    fun open(
        profile: ProxyProfile,
        destinationHost: String,
        destinationPort: Int,
        connectTimeoutMs: Int = DEFAULT_CONNECT_TIMEOUT_MS,
        readTimeoutMs: Int = DEFAULT_READ_TIMEOUT_MS,
        protectSocket: ((Socket) -> Boolean)? = null
    ): Socket {
        profile.validate()?.let { throw IllegalArgumentException(it) }
        require(destinationHost.isNotBlank() && !destinationHost.any(Char::isWhitespace)) {
            "A valid destination hostname or IP is required."
        }
        require(destinationPort in 1..DESTINATION_PORT_MAX) {
            "Destination port must be between 1 and 65535."
        }
        require(connectTimeoutMs > 0 && readTimeoutMs > 0) {
            "Socket timeouts must be positive."
        }

        val socket = Socket()
        try {
            if (protectSocket != null && !protectSocket(socket)) {
                throw IOException("Could not protect the upstream socket from VPN routing.")
            }
            socket.connect(InetSocketAddress(profile.host, profile.port), connectTimeoutMs)
            socket.soTimeout = readTimeoutMs

            val input = socket.getInputStream()
            val output = socket.getOutputStream()
            when (profile.protocol) {
                ProxyProtocol.HTTP_CONNECT -> establishHttpConnect(
                    profile, destinationHost, destinationPort, input, output
                )
                ProxyProtocol.SOCKS5 -> establishSocks5(
                    profile, destinationHost, destinationPort, input, output
                )
            }
            // The timeout is only for connection setup/handshake. A live tunnel
            // must remain usable while idle, like a normal TCP socket.
            socket.soTimeout = 0
            return socket
        } catch (error: Exception) {
            try {
                socket.close()
            } catch (_: IOException) {
                // Preserve the original connection or handshake failure.
            }
            throw error
        }
    }

    private fun establishHttpConnect(
        profile: ProxyProfile,
        host: String,
        port: Int,
        input: InputStream,
        output: OutputStream
    ) {
        val authority = formatAuthority(host, port)
        val authHeader = if (!profile.username.isNullOrEmpty()) {
            val credentials = "${profile.username}:${profile.password.orEmpty()}"
            val encoded = Base64.getEncoder().encodeToString(credentials.toByteArray(StandardCharsets.UTF_8))
            "Proxy-Authorization: Basic $encoded\r\n"
        } else {
            ""
        }
        val request = "CONNECT $authority HTTP/1.1\r\n" +
            "Host: $authority\r\n" +
            authHeader +
            "Proxy-Connection: Keep-Alive\r\n\r\n"
        output.write(request.toByteArray(StandardCharsets.US_ASCII))
        output.flush()

        val response = readHttpHeaders(input)
        val statusLine = response.substringBefore("\r\n")
        val statusCode = statusLine.split(' ').getOrNull(1)?.toIntOrNull()
        when (statusCode) {
            200 -> Unit
            407 -> throw SecurityException("Proxy authentication required or credentials rejected (HTTP 407).")
            null -> throw IOException("Invalid HTTP proxy response.")
            else -> throw IOException("HTTP proxy rejected CONNECT with status $statusCode.")
        }
    }

    private fun establishSocks5(
        profile: ProxyProfile,
        host: String,
        port: Int,
        input: InputStream,
        output: OutputStream
    ) {
        val hasCredentials = !profile.username.isNullOrEmpty()
        output.write(
            if (hasCredentials) byteArrayOf(0x05, 0x02, 0x00, 0x02)
            else byteArrayOf(0x05, 0x01, 0x00)
        )
        output.flush()

        val version = readByte(input)
        val method = readByte(input)
        if (version != 5) throw IOException("Server is not a SOCKS5 proxy.")
        when (method) {
            0 -> Unit
            2 -> {
                if (!hasCredentials) {
                    throw SecurityException("SOCKS5 server requires a username and password.")
                }
                val username = profile.username!!.toByteArray(StandardCharsets.UTF_8)
                val password = profile.password.orEmpty().toByteArray(StandardCharsets.UTF_8)
                require(username.size in 1..255 && password.size in 1..255) {
                    "SOCKS5 credentials must be 1–255 bytes."
                }
                output.write(byteArrayOf(0x01, username.size.toByte()))
                output.write(username)
                output.write(password.size)
                output.write(password)
                output.flush()
                val authVersion = readByte(input)
                val authStatus = readByte(input)
                if (authVersion != 1 || authStatus != 0) {
                    throw SecurityException("SOCKS5 authentication failed.")
                }
            }
            0xFF -> throw SecurityException("SOCKS5 server rejected all authentication methods.")
            else -> throw IOException("Unsupported SOCKS5 authentication method.")
        }

        val address = encodeSocksAddress(host, profile.remoteDns)
        output.write(byteArrayOf(0x05, 0x01, 0x00) + address +
            byteArrayOf((port ushr 8).toByte(), port.toByte()))
        output.flush()

        val responseVersion = readByte(input)
        val reply = readByte(input)
        readByte(input) // Reserved
        val addressType = readByte(input)
        if (responseVersion != 5) throw IOException("Invalid SOCKS5 response.")
        if (reply != 0) throw IOException("SOCKS5 CONNECT failed (reply code $reply).")
        val boundAddressLength = when (addressType) {
            1 -> 4
            4 -> 16
            3 -> readByte(input)
            else -> throw IOException("Unknown SOCKS5 address type.")
        }
        skipExactly(input, boundAddressLength + 2)
    }

    private fun encodeSocksAddress(host: String, remoteDns: Boolean): ByteArray {
        if (remoteDns) {
            val bytes = host.toByteArray(StandardCharsets.US_ASCII)
            require(bytes.isNotEmpty() && bytes.size <= 255 && bytes.all { it.toInt() in 33..126 }) {
                "Remote DNS requires an ASCII hostname of 1–255 bytes."
            }
            return byteArrayOf(0x03, bytes.size.toByte()) + bytes
        }

        val address = InetAddress.getByName(host).address
        return if (address.size == 4) byteArrayOf(0x01) + address
        else byteArrayOf(0x04) + address
    }

    private fun formatAuthority(host: String, port: Int): String {
        val normalizedHost = if (host.startsWith("[") && host.endsWith("]")) {
            host
        } else if (host.contains(':')) {
            "[$host]"
        } else {
            host
        }
        return "$normalizedHost:$port"
    }

    private fun readHttpHeaders(input: InputStream): String {
        val bytes = ArrayList<Byte>()
        val terminator = byteArrayOf(13, 10, 13, 10)
        var matched = 0
        while (bytes.size < MAX_HTTP_HEADER_BYTES) {
            val next = input.read()
            if (next < 0) break
            bytes.add(next.toByte())
            if (next == (terminator[matched].toInt() and 0xff)) {
                matched++
                if (matched == terminator.size) break
            } else {
                matched = if (next == 13) 1 else 0
            }
        }
        if (matched != terminator.size) throw IOException("Incomplete HTTP proxy response.")
        return bytes.toByteArray().toString(StandardCharsets.US_ASCII)
    }

    private fun readByte(input: InputStream): Int {
        val value = input.read()
        if (value < 0) throw IOException("Unexpected end of proxy handshake.")
        return value
    }

    private fun skipExactly(input: InputStream, count: Int) {
        repeat(count) { readByte(input) }
    }
}

package com.aboezzat.vpn.proxy

import java.io.IOException
import java.net.Socket

/**
 * Owns the lifecycle of concurrently opened upstream TCP tunnels.
 *
 * Call release() when the consumer finishes with a socket. closeAll() closes
 * every currently registered tunnel but keeps the manager reusable.
 * close()/shutdown() permanently rejects new tunnels and closes active ones.
 *
 * This manager does not detect a remote half-close by itself; the relay owner
 * must call release() from its finally block when either side finishes.
 */
class ProxyTcpTunnelManager(
    private val maxConcurrentTunnels: Int = DEFAULT_MAX_CONCURRENT_TUNNELS
) : AutoCloseable {
    private val lock = Any()
    private val activeTunnels = LinkedHashSet<Socket>()
    private var openings = 0
    private var closed = false

    init {
        require(maxConcurrentTunnels > 0) { "Maximum concurrent tunnels must be positive." }
    }

    val activeCount: Int
        get() = synchronized(lock) { activeTunnels.size }

    val isShutdown: Boolean
        get() = synchronized(lock) { closed }

    @Throws(IOException::class)
    fun openTunnel(
        profile: ProxyProfile,
        destinationHost: String,
        destinationPort: Int,
        connectTimeoutMs: Int = 7_000,
        readTimeoutMs: Int = 7_000,
        protectSocket: ((Socket) -> Boolean)? = null
    ): Socket {
        synchronized(lock) {
            if (closed) throw IOException("TCP tunnel manager is shut down.")
            if (activeTunnels.size + openings >= maxConcurrentTunnels) {
                throw IOException("Maximum concurrent TCP tunnel limit reached.")
            }
            openings++
        }

        val socket = try {
            ProxyTcpTunnel.open(
                profile = profile,
                destinationHost = destinationHost,
                destinationPort = destinationPort,
                connectTimeoutMs = connectTimeoutMs,
                readTimeoutMs = readTimeoutMs,
                protectSocket = protectSocket
            )
        } catch (error: Exception) {
            synchronized(lock) { openings-- }
            throw error
        }

        val accepted = synchronized(lock) {
            openings--
            if (closed) false else activeTunnels.add(socket)
        }
        if (!accepted) {
            try {
                socket.close()
            } catch (_: IOException) {
                // Preserve the manager shutdown result.
            }
            throw IOException("TCP tunnel manager was shut down while the tunnel was opening.")
        }
        return socket
    }

    /** Removes and closes a tunnel. Safe to call more than once. */
    fun release(socket: Socket): Boolean {
        val wasManaged = synchronized(lock) { activeTunnels.remove(socket) }
        if (wasManaged) {
            try {
                socket.close()
            } catch (_: IOException) {
                // The tunnel is removed from ownership even if close reports an error.
            }
        }
        return wasManaged
    }

    /** Closes all active tunnels but allows subsequent openTunnel() calls. */
    fun closeAll() {
        val sockets = synchronized(lock) {
            activeTunnels.toList().also { activeTunnels.clear() }
        }
        sockets.forEach { socket ->
            try {
                socket.close()
            } catch (_: IOException) {
                // Continue closing the remaining tunnels.
            }
        }
    }

    /** Permanently shuts down this manager and closes every active tunnel. */
    fun shutdown() {
        synchronized(lock) { closed = true }
        closeAll()
    }

    override fun close() = shutdown()

    companion object {
        const val DEFAULT_MAX_CONCURRENT_TUNNELS = 64
    }
}

/**
 * Verifies SOCKS5 UDP support with a real DNS datagram round-trip.
 * The TCP control connection remains open for the lifetime of the UDP association.
 */
data class ProxyUdpTestResult(
    val supported: Boolean,
    val message: String,
    val relayAddress: String? = null,
    val relayPort: Int? = null
)

object Socks5UdpAssociateTester {
    private const val CONNECT_TIMEOUT_MS = 5000
    private const val READ_TIMEOUT_MS = 5000
    private const val DNS_TRANSACTION_ID = 0xA1B2

    fun test(
        profile: ProxyProfile,
        protectSocket: ((Socket) -> Boolean)? = null,
        protectDatagramSocket: ((DatagramSocket) -> Boolean)? = null
    ): ProxyUdpTestResult {
        profile.validate()?.let { return ProxyUdpTestResult(false, it) }
        if (profile.protocol != ProxyProtocol.SOCKS5) {
            return ProxyUdpTestResult(
                false,
                "UDP is not available through HTTP CONNECT. This upstream mode carries TCP only."
            )
        }

        return try {
            Socket().use { control ->
                if (protectSocket != null && !protectSocket(control)) {
                    throw IOException("Could not protect the SOCKS5 control socket from VPN routing.")
                }
                control.connect(InetSocketAddress(profile.host, profile.port), CONNECT_TIMEOUT_MS)
                control.soTimeout = READ_TIMEOUT_MS
                val input = BufferedInputStream(control.getInputStream())
                val output = BufferedOutputStream(control.getOutputStream())
                negotiateAuthentication(profile, input, output)

                // Keep this TCP control socket open until the UDP relay test completes.
                output.write(byteArrayOf(0x05, 0x03, 0x00, 0x01, 0, 0, 0, 0, 0, 0))
                output.flush()

                if (input.read() != 5) throw IOException("Invalid SOCKS5 UDP ASSOCIATE response.")
                val reply = input.read()
                input.read() // RSV
                val addressType = input.read()
                if (reply != 0) throw IOException("SOCKS5 UDP ASSOCIATE rejected (reply code $reply).")

                val relayHost = when (addressType) {
                    1 -> InetAddress.getByAddress(readExactly(input, 4)).hostAddress
                    4 -> InetAddress.getByAddress(readExactly(input, 16)).hostAddress
                    3 -> {
                        val length = input.read()
                        if (length !in 1..255) throw IOException("Invalid SOCKS5 relay domain.")
                        String(readExactly(input, length), StandardCharsets.US_ASCII)
                    }
                    else -> throw IOException("Unknown SOCKS5 relay address type.")
                }
                val portBytes = readExactly(input, 2)
                val relayPort = ((portBytes[0].toInt() and 0xff) shl 8) or
                    (portBytes[1].toInt() and 0xff)
                if (relayPort == 0) throw IOException("SOCKS5 server returned an invalid UDP relay port.")

                DatagramSocket(null).use { datagram ->
                    // Bind first so Android has a real socket fd to protect;
                    // protect still runs before any packet is sent.
                    datagram.reuseAddress = true
                    datagram.bind(InetSocketAddress(0))
                    if (protectDatagramSocket != null && !protectDatagramSocket(datagram)) {
                        throw IOException("Could not protect the UDP datagram socket from VPN routing.")
                    }
                    datagram.soTimeout = READ_TIMEOUT_MS

                    val relay = InetSocketAddress(InetAddress.getByName(relayHost), relayPort)
                    val dnsQuery = buildDnsQuery()
                    val request = buildSocksUdpRequest(dnsQuery)
                    datagram.send(DatagramPacket(request, request.size, relay))

                    val responseBuffer = ByteArray(2048)
                    val response = DatagramPacket(responseBuffer, responseBuffer.size)
                    datagram.receive(response)
                    val dnsPayload = extractDnsPayload(response.data.copyOf(response.length))
                    if (dnsPayload.size < 12) throw IOException("UDP relay returned an incomplete DNS response.")
                    val transactionId = ((dnsPayload[0].toInt() and 0xff) shl 8) or
                        (dnsPayload[1].toInt() and 0xff)
                    if (transactionId != DNS_TRANSACTION_ID || (dnsPayload[2].toInt() and 0x80) == 0) {
                        throw IOException("UDP relay response did not match the DNS test request.")
                    }
                }

                ProxyUdpTestResult(
                    true,
                    "SOCKS5 UDP datagram round-trip verified with a DNS response.",
                    relayHost,
                    relayPort
                )
            }
        } catch (_: java.net.SocketTimeoutException) {
            ProxyUdpTestResult(false, "SOCKS5 UDP relay timed out: no DNS response was received.")
        } catch (error: Exception) {
            ProxyUdpTestResult(
                false,
                error.message?.take(140) ?: "SOCKS5 UDP datagram test failed."
            )
        }
    }

    private fun buildDnsQuery(): ByteArray {
        val name = byteArrayOf(
            7, 101, 120, 97, 109, 112, 108, 101,
            3, 99, 111, 109, 0
        )
        return byteArrayOf(
            (DNS_TRANSACTION_ID ushr 8).toByte(), DNS_TRANSACTION_ID.toByte(),
            0x01, 0x00, 0x00, 0x01, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00
        ) + name + byteArrayOf(0x00, 0x01, 0x00, 0x01)
    }

    private fun buildSocksUdpRequest(payload: ByteArray): ByteArray =
        byteArrayOf(0, 0, 0, 1, 1, 1, 1, 1, 0, 53) + payload

    private fun extractDnsPayload(packet: ByteArray): ByteArray {
        if (packet.size < 10 || packet[0].toInt() != 0 || packet[1].toInt() != 0) {
            throw IOException("Invalid SOCKS5 UDP relay response header.")
        }
        if (packet[2].toInt() != 0) throw IOException("Fragmented SOCKS5 UDP response is unsupported.")
        val addressLength = when (packet[3].toInt() and 0xff) {
            1 -> 4
            4 -> 16
            3 -> {
                if (packet.size < 5) throw IOException("Invalid UDP response domain header.")
                1 + (packet[4].toInt() and 0xff)
            }
            else -> throw IOException("Unknown UDP response address type.")
        }
        val payloadOffset = 4 + addressLength + 2
        if (packet.size < payloadOffset + 12) throw IOException("UDP relay response has no DNS payload.")
        return packet.copyOfRange(payloadOffset, packet.size)
    }

    private fun negotiateAuthentication(
        profile: ProxyProfile,
        input: BufferedInputStream,
        output: BufferedOutputStream
    ) {
        val hasCredentials = !profile.username.isNullOrEmpty()
        output.write(if (hasCredentials) byteArrayOf(5, 2, 0, 2) else byteArrayOf(5, 1, 0))
        output.flush()
        val version = input.read()
        val method = input.read()
        if (version != 5) throw IOException("Server is not a SOCKS5 proxy.")
        when (method) {
            0 -> Unit
            2 -> {
                if (!hasCredentials) throw IOException("SOCKS5 server requires username and password.")
                val username = profile.username!!.toByteArray(StandardCharsets.UTF_8)
                val password = profile.password.orEmpty().toByteArray(StandardCharsets.UTF_8)
                require(username.size in 1..255 && password.size in 1..255) {
                    "SOCKS5 credentials must be 1–255 bytes."
                }
                output.write(byteArrayOf(1, username.size.toByte()))
                output.write(username)
                output.write(password.size)
                output.write(password)
                output.flush()
                if (input.read() != 1 || input.read() != 0) throw IOException("SOCKS5 authentication failed.")
            }
            0xff -> throw IOException("SOCKS5 server rejected all authentication methods.")
            else -> throw IOException("Unsupported SOCKS5 authentication method.")
        }
    }

    private fun readExactly(input: BufferedInputStream, count: Int): ByteArray {
        val result = ByteArray(count)
        for (index in 0 until count) {
            val value = input.read()
            if (value < 0) throw IOException("Incomplete SOCKS5 UDP ASSOCIATE response.")
            result[index] = value.toByte()
        }
        return result
    }

import java.io.IOException
import java.net.Socket
import java.util.concurrent.ExecutorService
import java.util.concurrent.Future
import java.util.concurrent.atomic.AtomicBoolean
import java.util.concurrent.atomic.AtomicInteger

/**
 * Relays bytes bidirectionally between an accepted client TCP socket and an
 * already-established upstream proxy tunnel.
 *
 * The caller owns the executor and must provide a bounded, long-lived pool.
 * The session owns both sockets after start() succeeds and closes them when
 * both directions finish, either direction fails, or close() is called.
 *
 * This is a stream relay primitive only: it does not parse IP/TCP packets,
 * create a TUN interface, or implement TCP state tracking for VPN packets.
 */
object TcpStreamRelay {
    fun start(
        client: Socket,
        upstream: Socket,
        executor: ExecutorService,
        onClosed: (() -> Unit)? = null
    ): Session {
        require(client !== upstream) { "Client and upstream sockets must be different." }
        require(!client.isClosed && !upstream.isClosed) { "Both sockets must be open." }

        val session = Session(client, upstream, onClosed)
        try {
            session.submit(executor, client, upstream)
            session.submit(executor, upstream, client)
        } catch (error: Exception) {
            session.close()
            throw error
        }
        return session
    }

    class Session internal constructor(
        private val client: Socket,
        private val upstream: Socket,
        private val onClosed: (() -> Unit)?
    ) : AutoCloseable {
        private val closed = AtomicBoolean(false)
        private val completedDirections = AtomicInteger(0)
        @Volatile private var first: Future<*>? = null
        @Volatile private var second: Future<*>? = null

        internal fun submit(executor: ExecutorService, source: Socket, destination: Socket) {
            val task = executor.submit {
                try {
                    // Do not close the input stream here: SocketInputStream.close()
                    // closes the whole socket and would break the reverse half-stream.
                    val input = source.getInputStream()
                    val output = destination.getOutputStream()
                    val buffer = ByteArray(DEFAULT_BUFFER_SIZE)
                    while (!closed.get()) {
                        val count = input.read(buffer)
                        if (count < 0) break
                        if (count > 0) {
                            output.write(buffer, 0, count)
                            output.flush()
                        }
                    }
                    try {
                        destination.shutdownOutput()
                    } catch (_: IOException) {
                        // The peer may already have closed the connection.
                    }
                    if (completedDirections.incrementAndGet() == 2) close()
                } catch (_: IOException) {
                    close()
                } catch (_: RuntimeException) {
                    close()
                }
            }
            synchronized(this) {
                if (first == null) first = task else second = task
                if (closed.get()) task.cancel(true)
            }
        }

        override fun close() {
            if (!closed.compareAndSet(false, true)) return
            try { client.close() } catch (_: IOException) { }
            try { upstream.close() } catch (_: IOException) { }
            first?.cancel(true)
            second?.cancel(true)
            try { onClosed?.invoke() } catch (_: RuntimeException) { }
        }

        val isClosed: Boolean
            get() = closed.get()
    }
}

                    val buffer = ByteArray(64 * 1024)

                        val count = input.read(buffer)
                        if (count < 0) break
                        fileOutput.write(buffer, 0, count)
                        downloaded += count
                        onProgress(downloaded, total)
                    }
                    fileOutput.fd.sync()
                }
            }

                throw IllegalStateException("Downloaded APK size is incomplete.")
            }

            if (!temporary.renameTo(apk)) throw IllegalStateException("Could not finalize downloaded APK.")
            return apk
        } catch (error: Exception) {
            temporary.delete()
            throw error
        } finally {
            connection.disconnect()
        }
    }

    private fun startTunnel(intent: Intent, startId: Int) {
        trafficObservedLogged = false
        val receiver = intent.getParcelableExtra<ResultReceiver>(EXTRA_RESULT_RECEIVER)
        if (tunnelRunner.isActive()) {
            receiver?.send(RESULT_ALREADY_ACTIVE, Bundle().apply {
                putBoolean(EXTRA_SUCCESS, true)
                putString(EXTRA_MESSAGE, "A tunnel session is already active.")
            })
            return
        }
        if (!nativeBridgeBound) {
            failStart(receiver, startId, "Native tunnel engine could not be loaded.")
            return
        }
        if (VpnService.prepare(this) != null) {
            failStart(receiver, startId, "Android VPN permission is required.")
            return
        }
        try {
            startTunnelForeground()
        } catch (error: Exception) {
            failStart(
                receiver,
                startId,
                error.message ?: "Android did not allow the VPN foreground service to start."
            )
            return
        }
        val profile = parseProfile(intent)
        if (profile == null) {
            failStart(receiver, startId, "Invalid proxy profile.")
            return
        }

        // Resolve the proxy endpoint before creating the TUN route. The native
        // engine receives a numeric address, preventing a DNS bootstrap loop.
        val resolvedHost = resolveProxyHost(profile.host)
        if (resolvedHost == null) {
            failStart(receiver, startId, "Could not resolve the proxy server address.")
            return
        }

        val tun = try {
            establishTunInterface()
        } catch (error: Exception) {
            null
        }
        if (tun == null) {
            failStart(receiver, startId, "Android could not establish the TUN interface.")
            return
        }

        val command = Tun2ProxyCommandBuilder.build(
            profile.copy(host = resolvedHost),
            tunFd = tun.fd,
            mtu = TUN_MTU
        )
        if (command == null) {
            runCatching { tun.close() }
            failStart(receiver, startId, "The selected proxy profile is not valid for the tunnel.")
            return
        }

        val nativeArmed = try {
            NativeSocketProtectorBridge.nativeArmEngineStart()
        } catch (_: UnsatisfiedLinkError) {
            false
        }
        if (!nativeArmed) {
            runCatching { tun.close() }
            failStart(receiver, startId, "The native tunnel engine is already starting or unavailable.")
            return
        }

        activeReceiver = receiver
        stopRequested = false
        val accepted = tunnelRunner.start(
            cliArgs = command,
            mtu = TUN_MTU,
            packetInformation = false,
            closeTunDescriptor = { tun.close() }
        )
        if (!accepted) {
            activeReceiver = null
            tunnelRunning = false
            runCatching { NativeSocketProtectorBridge.nativeCancelEngineStart() }
            runCatching { tun.close() }
            failStart(receiver, startId, "The native tunnel worker is not available.")
            return
        }

        // Only report the tunnel as running after the native worker accepted the session.
        tunnelRunning = true
        ConnectionDiagnostics.add(this, "Tunnel worker accepted; upstream internet traffic is not yet verified; direct bypass apps=" + directBypassCount)
        receiver?.send(
            RESULT_STARTED,
            Bundle().apply {
                putBoolean(EXTRA_SUCCESS, true)
                putInt(EXTRA_DIRECT_BYPASS_COUNT, directBypassCount)
                putString(EXTRA_MESSAGE, "Tunnel worker started via ${profile.name}. Direct bypass active for $directBypassCount app(s); other saved route modes are not separately enforced yet.")
            }
        )
        updateTunnelNotification("Tunnel worker running via ${profile.name} · internet not verified · $directBypassCount direct bypass")
    }

    private fun stopTunnel(intent: Intent, startId: Int) {
        val receiver = intent.getParcelableExtra<ResultReceiver>(EXTRA_RESULT_RECEIVER)
        if (!tunnelRunner.isActive()) {
            receiver?.send(RESULT_STOPPED, Bundle().apply {
                putBoolean(EXTRA_SUCCESS, true)
                putString(EXTRA_MESSAGE, "Tunnel is already stopped.")
            })
            stopForeground(STOP_FOREGROUND_REMOVE)
            stopSelfResult(startId)
            return
        }

        stopRequested = true
        updateTunnelNotification("Stopping the tunnel and releasing resources")
        val accepted = tunnelRunner.stop()
        receiver?.send(
            if (accepted) RESULT_STOPPING else RESULT_STOP_FAILED,
            Bundle().apply {
                putBoolean(EXTRA_SUCCESS, accepted)
                putString(
                    EXTRA_MESSAGE,
                    if (accepted) "Tunnel stop requested. Waiting for native cleanup."
                    else "The native engine did not accept the stop request."
                )
            }
        )
    }

    private fun onNativeTunnelFinished(result: Int) {
        val receiver = activeReceiver
        val wasRequestedStop = stopRequested
        val successfulStop = result == 0 || (result == -13 && wasRequestedStop)
        activeReceiver = null
        stopRequested = false
        tunnelRunning = false

        if (destroyed) unbindNativeBridge()

        mainHandler.post {
            if (!destroyed) {
                val message = when {
                    successfulStop && wasRequestedStop -> "Tunnel stopped and native resources released."
                    successfulStop -> "Tunnel engine exited."
                    else -> "Native tunnel stopped with error code $result."
                }
                receiver?.send(
                    if (successfulStop) RESULT_STOPPED else RESULT_FAILED,
                    Bundle().apply {
                        putBoolean(EXTRA_SUCCESS, successfulStop)
                        putString(EXTRA_MESSAGE, message)
                    }
                )
                stopForeground(STOP_FOREGROUND_REMOVE)
                stopSelfResult(latestStartId)
            }
        }
    }

    private fun failStart(receiver: ResultReceiver?, startId: Int, message: String) {
        receiver?.send(RESULT_FAILED, Bundle().apply {
            putBoolean(EXTRA_SUCCESS, false)
            putString(EXTRA_MESSAGE, message)
        })
        stopForeground(STOP_FOREGROUND_REMOVE)
        stopSelfResult(startId)
    }

    private fun establishTunInterface(): ParcelFileDescriptor? {
        val routes = AppRouteStore(this).load()
        val directCandidates = routes.filterValues { it == RouteMode.DIRECT }.keys
        val safeDirectPackages = directCandidates.filter { candidatePackage ->
            if (candidatePackage == packageName) return@filter false
            runCatching {
                val uid = packageManager.getApplicationInfo(candidatePackage, 0).uid
                val uidPackages = packageManager.getPackagesForUid(uid)?.toList().orEmpty()
                uidPackages.isNotEmpty() && uidPackages.all { routes[it] == RouteMode.DIRECT }
            }.getOrDefault(false)
        }

        val builder = Builder()
            .setSession("AboEzzat VPN")
            .setMtu(TUN_MTU)
            .addAddress(InetAddress.getByName(IPV4_TUN_ADDRESS), 32)
            .addRoute(InetAddress.getByName("0.0.0.0"), 0)
            .addDnsServer(InetAddress.getByName(IPV4_DNS_ADDRESS))
            .addAddress(InetAddress.getByName(IPV6_TUN_ADDRESS), 128)
            .addRoute(InetAddress.getByName("::"), 0)
            .addDnsServer(InetAddress.getByName(IPV6_DNS_ADDRESS))
            .setBlocking(false)

        var applied = 0
        safeDirectPackages.forEach { candidatePackage ->
            runCatching {
                builder.addDisallowedApplication(candidatePackage)
                applied++
            }
        }
        directBypassCount = applied
        return builder.establish()
    }

    private fun resolveProxyHost(host: String): String? {
        val normalized = host.removePrefix("[").removeSuffix("]")
        if (normalized.contains(':') || normalized.matches(Regex("[0-9.]+"))) {
            return normalized
        }

        return try {
            val connectivity = getSystemService(ConnectivityManager::class.java)
            val network = connectivity.activeNetwork
            val addresses = if (network != null) {
                network.getAllByName(normalized)
            } else {
                InetAddress.getAllByName(normalized)
            }
            addresses.firstOrNull { it is Inet4Address }?.hostAddress
                ?: addresses.firstOrNull()?.hostAddress?.substringBefore('%')
        } catch (_: Exception) {
            null
        }
    }

    }

package com.aboezzat.vpn.vpn

import android.content.Context
import android.net.ConnectivityManager
import android.os.Build
import android.os.Process
import android.system.OsConstants
import java.net.InetSocketAddress

enum class IpTransportProtocol {
    TCP,
    UDP
}

data class NetworkConnectionTuple(
    val protocol: IpTransportProtocol,
    val local: InetSocketAddress,
    val remote: InetSocketAddress
)

sealed interface ConnectionOwnerUidResult {
    data class Found(val uid: Int) : ConnectionOwnerUidResult
    data object UnsupportedAndroidVersion : ConnectionOwnerUidResult
    data object ConnectionNotFound : ConnectionOwnerUidResult
    data object VpnNotActiveOrNotAuthorized : ConnectionOwnerUidResult
    data object InvalidConnectionTuple : ConnectionOwnerUidResult
    data class LookupFailed(val errorType: String) : ConnectionOwnerUidResult
}

/**
 * Resolves the UID owning an observed TCP/UDP connection.
 *
 * Android only exposes this API from API 29 and only to the active VpnService
 * for connections covered by that VPN. It must not be used as a guess when
 * the platform cannot provide an owner UID.
 */
class AndroidConnectionOwnerUidResolver(context: Context) : ConnectionOwnerUidProvider {
    private val connectivityManager =
        context.applicationContext.getSystemService(ConnectivityManager::class.java)

    override fun resolve(tuple: NetworkConnectionTuple): ConnectionOwnerUidResult {
        if (Build.VERSION.SDK_INT < Build.VERSION_CODES.Q) {
            return ConnectionOwnerUidResult.UnsupportedAndroidVersion
        }
        if (tuple.local.isUnresolved || tuple.remote.isUnresolved ||
            tuple.local.port !in 1..65535 || tuple.remote.port !in 1..65535
        ) {
            return ConnectionOwnerUidResult.InvalidConnectionTuple
        }

        val manager = connectivityManager
            ?: return ConnectionOwnerUidResult.LookupFailed("ConnectivityManagerUnavailable")

        val protocol = when (tuple.protocol) {
            IpTransportProtocol.TCP -> OsConstants.IPPROTO_TCP
            IpTransportProtocol.UDP -> OsConstants.IPPROTO_UDP
        }

        return try {
            val uid = manager.getConnectionOwnerUid(protocol, tuple.local, tuple.remote)
            if (uid == Process.INVALID_UID) {
                ConnectionOwnerUidResult.ConnectionNotFound
            } else {
                ConnectionOwnerUidResult.Found(uid)
            }
        } catch (_: SecurityException) {
            ConnectionOwnerUidResult.VpnNotActiveOrNotAuthorized
        } catch (_: IllegalArgumentException) {
            ConnectionOwnerUidResult.InvalidConnectionTuple
        } catch (error: Exception) {
            ConnectionOwnerUidResult.LookupFailed(error.javaClass.simpleName)
        }
    }
}

package com.aboezzat.vpn.vpn

import com.aboezzat.vpn.core.AppRoutePolicyResolver
import com.aboezzat.vpn.core.AppRouteResolution
import com.aboezzat.vpn.core.RouteMode

fun interface ConnectionOwnerUidProvider {
    fun resolve(tuple: NetworkConnectionTuple): ConnectionOwnerUidResult
}

sealed interface ConnectionRouteDecision {
    data class Resolved(
        val ownerUid: Int,
        val policy: AppRouteResolution
    ) : ConnectionRouteDecision

    data class Unresolved(
        val ownerLookup: ConnectionOwnerUidResult
    ) : ConnectionRouteDecision
}

/**
 * Joins Android connection attribution with the package-to-route policy map.
 * Unknown UIDs are never guessed: the caller receives Unresolved and must
 * apply its explicit fail-closed handling rather than silently misroute.
 */
class ConnectionRouteDecisionEngine(
    private val ownerUidProvider: ConnectionOwnerUidProvider,
    private val policyResolver: AppRoutePolicyResolver
) {
    fun decide(
        tuple: NetworkConnectionTuple,
        routes: Map<String, RouteMode>
    ): ConnectionRouteDecision {
        return when (val owner = ownerUidProvider.resolve(tuple)) {
            is ConnectionOwnerUidResult.Found -> {
                if (owner.uid <= 0) {
                    ConnectionRouteDecision.Unresolved(
                        ConnectionOwnerUidResult.ConnectionNotFound
                    )
                } else {
                    ConnectionRouteDecision.Resolved(
                        ownerUid = owner.uid,
                        policy = policyResolver.resolve(owner.uid, routes)
                    )
                }
            }
            else -> ConnectionRouteDecision.Unresolved(owner)
        }
    }
}

/**
 * DNS-over-TLS (RFC 7858) through the configured upstream proxy.
 *
 * Uses the platform trust store and hostname verification. No certificate
 * verification is disabled and no resolver is selected implicitly.
 */
class DnsOverTlsForwarder(
    private val connectTimeoutMs: Int = DEFAULT_TIMEOUT_MS,
    private val readTimeoutMs: Int = DEFAULT_TIMEOUT_MS,
    private val maxMessageBytes: Int = DnsTcpForwarder.MAX_DNS_MESSAGE_BYTES,
    private val socketFactory: SSLSocketFactory =
        SSLSocketFactory.getDefault() as SSLSocketFactory
) : DnsRequestForwarder {
    init {
        require(connectTimeoutMs > 0 && readTimeoutMs > 0)
        require(maxMessageBytes in DnsTcpForwarder.DNS_HEADER_BYTES..DnsTcpForwarder.MAX_DNS_MESSAGE_BYTES)
    }

    fun forward(
        profile: ProxyProfile,
        resolverHost: String,
        query: ByteArray
    ): ByteArray = forward(profile, resolverHost, query, null)

    @Throws(IOException::class)
    override fun forward(
        profile: ProxyProfile,
        resolverHost: String,
        query: ByteArray,
        protectSocket: ((Socket) -> Boolean)?
    ): ByteArray {
        require(resolverHost.isNotBlank() && !resolverHost.any(Char::isWhitespace)) {
            "A DNS-over-TLS resolver hostname or IP is required."
        }
        DnsTcpForwarder.validateQuery(query, maxMessageBytes)

        val tunnel = ProxyTcpTunnel.open(
            profile = profile,
            destinationHost = resolverHost,
            destinationPort = DNS_OVER_TLS_PORT,
            connectTimeoutMs = connectTimeoutMs,
            readTimeoutMs = readTimeoutMs,
            protectSocket = protectSocket
        )
        tunnel.use { rawSocket ->
            val tls = socketFactory.createSocket(
                rawSocket,
                resolverHost,
                DNS_OVER_TLS_PORT,
                true
            ) as SSLSocket
            tls.use { secureSocket ->
                secureSocket.soTimeout = readTimeoutMs
                val parameters = secureSocket.sslParameters
                parameters.endpointIdentificationAlgorithm = "HTTPS"
                if (!isIpLiteral(resolverHost)) {
                    parameters.serverNames = listOf(SNIHostName(resolverHost))
                }
                secureSocket.sslParameters = parameters
                secureSocket.startHandshake()

                val input = DataInputStream(secureSocket.getInputStream())
                val output = DataOutputStream(secureSocket.getOutputStream())
                output.writeShort(query.size)
                output.write(query)
                output.flush()

                val responseLength = input.readUnsignedShort()
                if (responseLength !in DnsTcpForwarder.DNS_HEADER_BYTES..maxMessageBytes) {
                    throw IOException("DNS-over-TLS resolver returned an invalid message length.")
                }
                val response = ByteArray(responseLength)
                input.readFully(response)
                DnsTcpForwarder.validateResponse(query, response)
                return response
            }
        }
    }

    private fun isIpLiteral(host: String): Boolean =
        host.contains(':') || IPV4_PATTERN.matches(host)

    companion object {
        const val DNS_OVER_TLS_PORT = 853
        const val DEFAULT_TIMEOUT_MS = 7_000
        private val IPV4_PATTERN =
            Regex("^(?:[0-9]{1,3}\\.){3}[0-9]{1,3}$")
    }
}

/**
 * Strict parser for a single-question DNS query with optional EDNS(0) OPT.
 *
 * It extracts routing metadata only. It does not resolve names, forward DNS,
 * synthesize replies, or parse DNS-over-TCP length prefixes.
 */
object DnsQueryParser {
    data class Question(
        val transactionId: Int,
        val name: String,
        val type: Int,
        val queryClass: Int,
        val recursionDesired: Boolean,
        val ednsPresent: Boolean = false,
        val udpPayloadSize: Int? = null,
        val ednsVersion: Int? = null,
        val dnssecOk: Boolean = false,
        val ednsExtendedRcode: Int? = null
    )

    fun parse(packet: ByteArray): Question? {
        if (packet.size < HEADER_SIZE + 5) return null
        val id = u16(packet, 0)
        val flags = u16(packet, 2)
        val questionCount = u16(packet, 4)
        val answerCount = u16(packet, 6)
        val authorityCount = u16(packet, 8)
        val additionalCount = u16(packet, 10)

        if ((flags and 0x8000) != 0) return null
        if (((flags ushr 11) and 0x0f) != 0) return null
        if (questionCount != 1 || answerCount != 0 || authorityCount != 0) return null
        if (additionalCount !in 0..1) return null

        var offset = HEADER_SIZE
        val labels = ArrayList<String>()
        var wireNameLength = 1 // terminating root label
        while (true) {
            if (offset >= packet.size) return null
            val length = packet[offset++].toInt() and 0xff
            if (length == 0) break
            if ((length and 0xc0) != 0 || length > 63) return null
            if (offset + length > packet.size) return null
            wireNameLength += length + 1
            if (wireNameLength > 255) return null
            val labelBytes = packet.copyOfRange(offset, offset + length)
            if (labelBytes.any { (it.toInt() and 0xff) !in 0x21..0x7e }) return null
            labels += labelBytes.toString(Charsets.US_ASCII)
            offset += length
        }
        if (labels.isEmpty() || offset + 4 > packet.size) return null

        val type = u16(packet, offset)
        val queryClass = u16(packet, offset + 2)
        offset += 4
        if (type == 0 || queryClass == 0) return null

        var ednsPresent = false
        var udpPayloadSize: Int? = null
        var ednsVersion: Int? = null
        var dnssecOk = false
        var ednsExtendedRcode: Int? = null

        if (additionalCount == 1) {
            // OPT must use the root owner name and be the only additional RR.
            if (offset + 11 > packet.size || packet[offset++].toInt() != 0) return null
            if (u16(packet, offset) != OPT_TYPE) return null
            offset += 2
            val payloadSize = u16(packet, offset)
            offset += 2
            if (payloadSize < 512) return null

            val ttlHigh = packet[offset++].toInt() and 0xff
            val version = packet[offset++].toInt() and 0xff
            val ttlFlags = u16(packet, offset)
            offset += 2
            val dataLength = u16(packet, offset)
            offset += 2
            if (offset + dataLength != packet.size) return null
            if (!validEdnsOptions(packet, offset, dataLength)) return null

            ednsPresent = true
            udpPayloadSize = payloadSize
            ednsVersion = version
            dnssecOk = (ttlFlags and 0x8000) != 0
            ednsExtendedRcode = ttlHigh
        } else if (offset != packet.size) {
            return null
        }

        return Question(
            transactionId = id,
            name = labels.joinToString(".").lowercase(),
            type = type,
            queryClass = queryClass,
            recursionDesired = (flags and 0x0100) != 0,
            ednsPresent = ednsPresent,
            udpPayloadSize = udpPayloadSize,
            ednsVersion = ednsVersion,
            dnssecOk = dnssecOk,
            ednsExtendedRcode = ednsExtendedRcode
        )
    }

    private fun validEdnsOptions(packet: ByteArray, start: Int, length: Int): Boolean {
        var offset = start
        val end = start + length
        while (offset < end) {
            if (offset + 4 > end) return false
            val optionLength = u16(packet, offset + 2)
            offset += 4
            if (offset + optionLength > end) return false
            offset += optionLength
        }
        return offset == end
    }

    private fun u16(data: ByteArray, offset: Int): Int =
        ((data[offset].toInt() and 0xff) shl 8) or
            (data[offset + 1].toInt() and 0xff)

    private const val HEADER_SIZE = 12
    private const val OPT_TYPE = 41
}

/**
 * Sends DNS over TCP through the selected upstream HTTP CONNECT or SOCKS5 proxy.
 * The resolver connection is not encrypted; use DnsOverTlsForwarder when the
 * selected resolver supports DNS-over-TLS.
 */
class DnsTcpForwarder(
    private val connectTimeoutMs: Int = DEFAULT_TIMEOUT_MS,
    private val readTimeoutMs: Int = DEFAULT_TIMEOUT_MS,
    private val maxMessageBytes: Int = MAX_DNS_MESSAGE_BYTES
) : DnsRequestForwarder {
    init {
        require(connectTimeoutMs > 0 && readTimeoutMs > 0)
        require(maxMessageBytes in DNS_HEADER_BYTES..MAX_DNS_MESSAGE_BYTES)
    }

    fun forward(
        profile: ProxyProfile,
        resolverHost: String,
        query: ByteArray
    ): ByteArray = forward(profile, resolverHost, query, null)

    @Throws(IOException::class)
    override fun forward(
        profile: ProxyProfile,
        resolverHost: String,
        query: ByteArray,
        protectSocket: ((Socket) -> Boolean)?
    ): ByteArray {
        validateQuery(query, maxMessageBytes)
        val socket = ProxyTcpTunnel.open(
            profile = profile,
            destinationHost = resolverHost,
            destinationPort = DNS_TCP_PORT,
            connectTimeoutMs = connectTimeoutMs,
            readTimeoutMs = readTimeoutMs,
            protectSocket = protectSocket
        )
        socket.use {
            it.soTimeout = readTimeoutMs
            val input = DataInputStream(it.getInputStream())
            val output = DataOutputStream(it.getOutputStream())
            output.writeShort(query.size)
            output.write(query)
            output.flush()
            val responseLength = input.readUnsignedShort()
            if (responseLength !in DNS_HEADER_BYTES..maxMessageBytes) {
                throw IOException("DNS resolver returned an invalid message length.")
            }
            val response = ByteArray(responseLength)
            input.readFully(response)
            validateResponse(query, response)
            return response
        }
    }

    companion object {
        const val DNS_TCP_PORT = 53
        const val DEFAULT_TIMEOUT_MS = 7_000
        const val DNS_HEADER_BYTES = 12
        const val MAX_DNS_MESSAGE_BYTES = 65_535

        internal fun validateQuery(query: ByteArray, maxMessageBytes: Int) {
            require(query.size in DNS_HEADER_BYTES..maxMessageBytes) {
                "DNS query length is outside the supported range."
            }
            require(DnsQueryParser.parse(query) != null) {
                "DNS query is malformed or uses an unsupported query layout."
            }
        }

        internal fun validateResponse(query: ByteArray, response: ByteArray) {
            if (response.size < DNS_HEADER_BYTES) throw IOException("DNS response is too short.")
            val queryId = readU16(query, 0)
            val responseId = readU16(response, 0)
            val flags = readU16(response, 2)
            if (responseId != queryId) throw IOException("DNS response transaction ID mismatch.")
            if ((flags and 0x8000) == 0) throw IOException("DNS upstream returned a query, not a response.")
            if (((flags ushr 11) and 0x0f) != 0) throw IOException("DNS response opcode mismatch.")
        }

        private fun readU16(bytes: ByteArray, offset: Int): Int =
            ((bytes[offset].toInt() and 0xff) shl 8) or
                (bytes[offset + 1].toInt() and 0xff)
    }
}

/**
 * Localhost-only UDP-to-TCP DNS adapter.
 *
 * A future TUN packet engine may send intercepted DNS datagrams to this
 * loopback endpoint. This class does not capture device traffic or establish
 * a VPN interface by itself.
 */
class DnsUdpTcpBridge(
    private val forwarder: DnsRequestForwarder = DnsTcpForwarder(),
    private val maxConcurrentRequests: Int = DEFAULT_MAX_CONCURRENT_REQUESTS,
    private val queueCapacity: Int = DEFAULT_QUEUE_CAPACITY
) : AutoCloseable {
    private val lock = Any()
    private val received = AtomicLong()
    private val forwarded = AtomicLong()
    private val dropped = AtomicLong()

    @Volatile private var datagramSocket: DatagramSocket? = null
    @Volatile private var worker: Thread? = null
    @Volatile private var executor: ThreadPoolExecutor? = null
    @Volatile private var running = false

    data class Stats(
        val received: Long,
        val forwarded: Long,
        val dropped: Long
    )

    init {
        require(maxConcurrentRequests > 0)
        require(queueCapacity > 0)
    }

    val isRunning: Boolean
        get() = running

    fun stats(): Stats = Stats(received.get(), forwarded.get(), dropped.get())

    /**
     * Starts a DNS listener bound only to 127.0.0.1 on an ephemeral port.
     * The returned address is intended for an in-process packet/TUN adapter.
     */
    @Throws(IOException::class)
    fun start(
        profile: ProxyProfile,
        resolverHost: String,
        protectSocket: ((Socket) -> Boolean)? = null
    ): InetSocketAddress = synchronized(lock) {
        check(!running && datagramSocket == null) { "DNS UDP bridge is already running or still shutting down." }
        require(resolverHost.isNotBlank() && !resolverHost.any(Char::isWhitespace)) {
            "A resolver hostname or IP is required."
        }

        val socket = DatagramSocket(null)
        try {
            socket.reuseAddress = false
            socket.bind(InetSocketAddress(InetAddress.getByName(LOOPBACK_HOST), 0))
            val pool = ThreadPoolExecutor(
                maxConcurrentRequests,
                maxConcurrentRequests,
                0L,
                TimeUnit.MILLISECONDS,
                ArrayBlockingQueue(queueCapacity),
                daemonThreadFactory(),
                ThreadPoolExecutor.AbortPolicy()
            )
            datagramSocket = socket
            executor = pool
            running = true
            val receiveThread = Thread({
                receiveLoop(socket, pool, profile, resolverHost, protectSocket)
            }, "AboEzzat-DNS-UDP-Bridge").apply {
                isDaemon = true
            }
            worker = receiveThread
            receiveThread.start()
            InetSocketAddress(InetAddress.getByName(LOOPBACK_HOST), socket.localPort)
        } catch (error: Exception) {
            running = false
            datagramSocket = null
            executor?.shutdownNow()
            executor = null
            socket.close()
            if (error is IOException) throw error
            throw IOException("Could not start the local DNS bridge.", error)
        }
    }

    private fun receiveLoop(
        socket: DatagramSocket,
        pool: ThreadPoolExecutor,
        profile: ProxyProfile,
        resolverHost: String,
        protectSocket: ((Socket) -> Boolean)?
    ) {
        val buffer = ByteArray(MAX_UDP_PACKET_BYTES)
        try {
            while (running && !socket.isClosed) {
                val packet = DatagramPacket(buffer, buffer.size)
                try {
                    socket.receive(packet)
                } catch (_: SocketException) {
                    if (running) dropped.incrementAndGet()
                    break
                }
                received.incrementAndGet()
                val query = packet.data.copyOfRange(packet.offset, packet.offset + packet.length)
                val clientAddress = packet.socketAddress
                try {
                    pool.execute {
                        try {
                            val response = forwarder.forward(profile, resolverHost, query, protectSocket)
                            if (running && response.size <= MAX_UDP_PACKET_BYTES) {
                                val reply = DatagramPacket(response, response.size, clientAddress)
                                socket.send(reply)
                                forwarded.incrementAndGet()
                            } else {
                                dropped.incrementAndGet()
                            }
                        } catch (_: Exception) {
                            // DNS transport failures are counted and dropped; never leak
                            // the query onto the device's direct network as a fallback.
                            dropped.incrementAndGet()
                        }
                    }
                } catch (_: RejectedExecutionException) {
                    dropped.incrementAndGet()
                }
            }
        } finally {
            synchronized(lock) {
                if (datagramSocket === socket) {
                    running = false
                    datagramSocket = null
                    executor = null
                    worker = null
                }
            }
            pool.shutdownNow()
            socket.close()
        }
    }

    override fun close() {
        val socket: DatagramSocket?
        val pool: ThreadPoolExecutor?
        val thread: Thread?
        synchronized(lock) {
            running = false
            socket = datagramSocket
            pool = executor
            thread = worker
            datagramSocket = null
            executor = null
            worker = null
        }
        socket?.close()
        pool?.shutdownNow()
        if (thread != null && thread !== Thread.currentThread()) {
            try {
                thread.join(CLOSE_JOIN_TIMEOUT_MS)
            } catch (_: InterruptedException) {
                Thread.currentThread().interrupt()
            }
        }
    }

    private fun daemonThreadFactory() = ThreadFactory { runnable ->
        Thread(runnable, "AboEzzat-DNS-Forward").apply { isDaemon = true }
    }

    companion object {
        const val LOOPBACK_HOST = "127.0.0.1"
        const val MAX_UDP_PACKET_BYTES = 65_507
        const val DEFAULT_MAX_CONCURRENT_REQUESTS = 8
        const val DEFAULT_QUEUE_CAPACITY = 32
        const val CLOSE_JOIN_TIMEOUT_MS = 1_000L
    }
}

package com.aboezzat.vpn.vpn

import java.net.InetAddress

/**
 * Extracts the 5-tuple needed for Android's connection-owner UID lookup from
 * an outbound IPv4 TCP or UDP packet read from a future VpnService TUN.
 *
 * Fragmented packets are rejected because the transport ports may not exist
 * in every fragment. IPv6 and non-TCP/UDP protocols are not supported here.
 */
object Ipv4ConnectionTupleParser {
    fun parseOutbound(packet: ByteArray): NetworkConnectionTuple {
        require(packet.size >= IPV4_MIN_HEADER_BYTES) { "IPv4 packet is too short." }
        val versionAndIhl = u8(packet, 0)
        require((versionAndIhl ushr 4) == IPV4_VERSION) { "Only IPv4 is supported." }

        val ipHeaderLength = (versionAndIhl and 0x0f) * 4
        require(ipHeaderLength in IPV4_MIN_HEADER_BYTES..IPV4_MAX_HEADER_BYTES) {
            "Invalid IPv4 header length."
        }
        require(packet.size >= ipHeaderLength) { "IPv4 header is truncated." }

        val totalLength = u16(packet, 2)
        require(totalLength == packet.size && totalLength >= ipHeaderLength) {
            "IPv4 total length does not match packet size."
        }
        require(checksum(packet, 0, ipHeaderLength) == 0) {
            "Invalid IPv4 header checksum."
        }

        val fragmentField = u16(packet, 6)
        require((fragmentField and RESERVED_FLAG) == 0) {
            "IPv4 reserved fragment flag must be zero."
        }
        require((fragmentField and FRAGMENT_MASK) == 0) {
            "Fragmented IPv4 packets are not supported for UID lookup."
        }

        val protocol = when (u8(packet, 9)) {
            TCP_PROTOCOL -> IpTransportProtocol.TCP
            UDP_PROTOCOL -> IpTransportProtocol.UDP
            else -> throw IllegalArgumentException("Only TCP and UDP packets can be attributed.")
        }

        val transportOffset = ipHeaderLength
        val transportLength = totalLength - ipHeaderLength
        require(transportLength >= if (protocol == IpTransportProtocol.TCP) TCP_MIN_HEADER_BYTES else UDP_HEADER_BYTES) {
            "Transport header is truncated."
        }

        val sourcePort = u16(packet, transportOffset)
        val destinationPort = u16(packet, transportOffset + 2)
        require(sourcePort in 1..65535 && destinationPort in 1..65535) {
            "Invalid transport port."
        }

        when (protocol) {
            IpTransportProtocol.TCP -> {
                val tcpHeaderLength = (u8(packet, transportOffset + 12) ushr 4) * 4
                require(tcpHeaderLength in TCP_MIN_HEADER_BYTES..transportLength) {
                    "Invalid TCP data offset."
                }
            }
            IpTransportProtocol.UDP -> {
                val udpLength = u16(packet, transportOffset + 4)
                require(udpLength == transportLength && udpLength >= UDP_HEADER_BYTES) {
                    "Invalid UDP length."
                }
            }
        }

        val localAddress = InetAddress.getByAddress(packet.copyOfRange(12, 16))
        val remoteAddress = InetAddress.getByAddress(packet.copyOfRange(16, 20))
        return NetworkConnectionTuple(
            protocol = protocol,
            local = java.net.InetSocketAddress(localAddress, sourcePort),
            remote = java.net.InetSocketAddress(remoteAddress, destinationPort)
        )
    }

    private fun checksum(bytes: ByteArray, offset: Int, length: Int): Int {
        var sum = 0L
        var index = 0
        while (index + 1 < length) {
            sum += ((bytes[offset + index].toInt() and 0xff) shl 8) or
                (bytes[offset + index + 1].toInt() and 0xff)
            index += 2
        }
        if (index < length) sum += (bytes[offset + index].toInt() and 0xff) shl 8
        while ((sum ushr 16) != 0L) sum = (sum and 0xffff) + (sum ushr 16)
        return sum.toInt().inv() and 0xffff
    }

    private fun u8(bytes: ByteArray, offset: Int): Int =
        bytes[offset].toInt() and 0xff

    private fun u16(bytes: ByteArray, offset: Int): Int =
        (u8(bytes, offset) shl 8) or u8(bytes, offset + 1)

    private const val IPV4_VERSION = 4
    private const val IPV4_MIN_HEADER_BYTES = 20
    private const val IPV4_MAX_HEADER_BYTES = 60
    private const val TCP_MIN_HEADER_BYTES = 20
    private const val UDP_HEADER_BYTES = 8
    private const val TCP_PROTOCOL = 6
    private const val UDP_PROTOCOL = 17
    private const val RESERVED_FLAG = 0x8000
    private const val FRAGMENT_MASK = 0x3fff
}

/**
 * Parses complete, unfragmented outbound IPv4/TCP packets for a future TCP
 * session engine. This is a packet decoder only: it does not create sockets,
 * acknowledge packets, or establish a VPN interface.
 */
object Ipv4TcpPacketParser {
    data class Packet(
        val tuple: NetworkConnectionTuple,
        val sequenceNumber: Long,
        val acknowledgementNumber: Long,
        val flags: Int,
        val windowSize: Int,
        val urgentPointer: Int,
        val ipHeaderLength: Int,
        val tcpHeaderLength: Int,
        val ipOptions: ByteArray,
        val tcpOptions: ByteArray,
        val payload: ByteArray
    ) {
        val isFin: Boolean get() = flags and FLAG_FIN != 0
        val isSyn: Boolean get() = flags and FLAG_SYN != 0
        val isRst: Boolean get() = flags and FLAG_RST != 0
        val isPsh: Boolean get() = flags and FLAG_PSH != 0
        val isAck: Boolean get() = flags and FLAG_ACK != 0
        val isUrg: Boolean get() = flags and FLAG_URG != 0
        val isEce: Boolean get() = flags and FLAG_ECE != 0
        val isCwr: Boolean get() = flags and FLAG_CWR != 0
        val isNs: Boolean get() = flags and FLAG_NS != 0
    }

    fun parse(packet: ByteArray): Packet {
        val tuple = Ipv4ConnectionTupleParser.parseOutbound(packet)
        require(tuple.protocol == IpTransportProtocol.TCP) { "IPv4 packet is not TCP." }

        val ipHeaderLength = (u8(packet, 0) and 0x0f) * 4
        val tcpOffset = ipHeaderLength
        val tcpLength = packet.size - tcpOffset
        val dataOffsetAndReserved = u8(packet, tcpOffset + 12)
        require((dataOffsetAndReserved and RESERVED_TCP_BITS) == 0) {
            "TCP reserved bits must be zero."
        }
        val tcpHeaderLength = (dataOffsetAndReserved ushr 4) * 4
        require(tcpHeaderLength in TCP_MIN_HEADER_BYTES..tcpLength) {
            "Invalid TCP data offset."
        }

        val flags = ((dataOffsetAndReserved and NS_MASK) shl 8) or u8(packet, tcpOffset + 13)
        val sourceAddress = packet.copyOfRange(12, 16)
        val destinationAddress = packet.copyOfRange(16, 20)
        require(validTcpChecksum(sourceAddress, destinationAddress, packet, tcpOffset, tcpLength)) {
            "Invalid TCP checksum."
        }

        val tcpOptions = packet.copyOfRange(tcpOffset + TCP_MIN_HEADER_BYTES, tcpOffset + tcpHeaderLength)
        validateTcpOptions(tcpOptions)

        return Packet(
            tuple = tuple,
            sequenceNumber = u32(packet, tcpOffset + 4),
            acknowledgementNumber = u32(packet, tcpOffset + 8),
            flags = flags,
            windowSize = u16(packet, tcpOffset + 14),
            urgentPointer = u16(packet, tcpOffset + 18),
            ipHeaderLength = ipHeaderLength,
            tcpHeaderLength = tcpHeaderLength,
            ipOptions = packet.copyOfRange(IPV4_MIN_HEADER_BYTES, ipHeaderLength),
            tcpOptions = tcpOptions,
            payload = packet.copyOfRange(tcpOffset + tcpHeaderLength, packet.size)
        )
    }

    private fun validateTcpOptions(options: ByteArray) {
        var index = 0
        while (index < options.size) {
            when (val kind = u8(options, index)) {
                OPTION_END -> return
                OPTION_NOP -> index += 1
                else -> {
                    require(index + 1 < options.size) { "Truncated TCP option length." }
                    val length = u8(options, index + 1)
                    require(length >= 2 && index + length <= options.size) {
                        "Invalid TCP option length for kind $kind."
                    }
                    index += length
                }
            }
        }
    }

    private fun validTcpChecksum(
        source: ByteArray,
        destination: ByteArray,
        packet: ByteArray,
        offset: Int,
        length: Int
    ): Boolean {
        var sum = 0L
        for (index in 0 until 4 step 2) {
            sum += ((source[index].toInt() and 0xff) shl 8) or (source[index + 1].toInt() and 0xff)
            sum += ((destination[index].toInt() and 0xff) shl 8) or (destination[index + 1].toInt() and 0xff)
        }
        sum += TCP_PROTOCOL
        sum += length
        sum = addWords(sum, packet, offset, length)
        return finishChecksum(sum) == 0
    }

    private fun addWords(initial: Long, bytes: ByteArray, offset: Int, length: Int): Long {
        var sum = initial
        var index = 0
        while (index + 1 < length) {
            sum += ((bytes[offset + index].toInt() and 0xff) shl 8) or
                (bytes[offset + index + 1].toInt() and 0xff)
            index += 2
        }
        if (index < length) sum += (bytes[offset + index].toInt() and 0xff) shl 8
        return sum
    }

    private fun finishChecksum(value: Long): Int {
        var sum = value
        while ((sum ushr 16) != 0L) sum = (sum and 0xffff) + (sum ushr 16)
        return sum.toInt().inv() and 0xffff
    }

    private fun u8(bytes: ByteArray, offset: Int): Int = bytes[offset].toInt() and 0xff

    private fun u16(bytes: ByteArray, offset: Int): Int =
        (u8(bytes, offset) shl 8) or u8(bytes, offset + 1)

    private fun u32(bytes: ByteArray, offset: Int): Long =
        ((u16(bytes, offset).toLong() shl 16) or u16(bytes, offset + 2).toLong()) and 0xffff_ffffL

    private const val IPV4_MIN_HEADER_BYTES = 20
    private const val TCP_MIN_HEADER_BYTES = 20
    private const val TCP_PROTOCOL = 6
    private const val RESERVED_TCP_BITS = 0x0e
    private const val NS_MASK = 0x01
    private const val OPTION_END = 0
    private const val OPTION_NOP = 1

    const val FLAG_FIN = 0x001
    const val FLAG_SYN = 0x002
    const val FLAG_RST = 0x004
    const val FLAG_PSH = 0x008
    const val FLAG_ACK = 0x010
    const val FLAG_URG = 0x020
    const val FLAG_ECE = 0x040
    const val FLAG_CWR = 0x080
    const val FLAG_NS = 0x100

package com.aboezzat.vpn.vpn


/**
 * Strict IPv4/UDP DNS packet codec for a future TUN packet loop.
 *
 * It intentionally rejects fragments and non-IPv4 traffic. It does not open
 * sockets, capture traffic, or establish a VPN interface.
 */
object Ipv4UdpDnsPacketCodec {
    data class Query(
        val sourceAddress: ByteArray,
        val destinationAddress: ByteArray,
        val sourcePort: Int,
        val destinationPort: Int,
        val identification: Int,
        val dscpEcn: Int,
        val dontFragment: Boolean,
        val ttl: Int,
        val ipOptions: ByteArray,
        val dnsPayload: ByteArray
    )

    fun parseQuery(packet: ByteArray): Query {
        require(packet.size >= MIN_PACKET_BYTES) { "IPv4 UDP packet is too short." }
        val versionAndIhl = u8(packet, 0)
        require((versionAndIhl ushr 4) == IPV4_VERSION) { "Only IPv4 packets are supported." }
        val headerLength = (versionAndIhl and 0x0f) * 4
        require(headerLength in IPV4_MIN_HEADER_BYTES..IPV4_MAX_HEADER_BYTES) {
            "Invalid IPv4 header length."
        }
        require(packet.size >= headerLength + UDP_HEADER_BYTES) { "IPv4 packet is truncated." }

        val totalLength = u16(packet, 2)
        require(totalLength == packet.size && totalLength >= headerLength + UDP_HEADER_BYTES) {
            "IPv4 total length does not match the packet."
        }
        require(checksum(packet, 0, headerLength) == 0) { "Invalid IPv4 header checksum." }
        require(u8(packet, 9) == UDP_PROTOCOL) { "IPv4 packet is not UDP." }

        val fragmentField = u16(packet, 6)
        require((fragmentField and RESERVED_FLAG) == 0) {
            "IPv4 reserved fragment flag must be zero."
        }
        require((fragmentField and FRAGMENT_MASK) == 0) {
            "Fragmented IPv4 UDP packets are not supported."
        }

        val udpOffset = headerLength
        val udpLength = u16(packet, udpOffset + 4)
        require(udpLength >= UDP_HEADER_BYTES && headerLength + udpLength == totalLength) {
            "Invalid UDP length."
        }
        val sourcePort = u16(packet, udpOffset)
        val destinationPort = u16(packet, udpOffset + 2)
        require(destinationPort == DNS_PORT) { "UDP destination is not the DNS port." }

        val sourceAddress = packet.copyOfRange(12, 16)
        val destinationAddress = packet.copyOfRange(16, 20)
        val udpChecksum = u16(packet, udpOffset + 6)
        if (udpChecksum != 0) {
            require(verifyUdpChecksum(sourceAddress, destinationAddress, packet, udpOffset, udpLength)) {
                "Invalid UDP checksum."
            }
        }

        val dns = packet.copyOfRange(udpOffset + UDP_HEADER_BYTES, totalLength)
        require(DnsQueryParser.parse(dns) != null) { "Malformed or unsupported DNS query." }

        return Query(
            sourceAddress = sourceAddress,
            destinationAddress = destinationAddress,
            sourcePort = sourcePort,
            destinationPort = destinationPort,
            identification = u16(packet, 4),
            dscpEcn = u8(packet, 1),
            dontFragment = (fragmentField and DONT_FRAGMENT_FLAG) != 0,
            ttl = u8(packet, 8),
            ipOptions = packet.copyOfRange(IPV4_MIN_HEADER_BYTES, headerLength),
            dnsPayload = dns
        )
    }

    fun buildResponse(query: Query, dnsResponse: ByteArray, ttl: Int = DEFAULT_TTL): ByteArray {
        require(ttl in 1..255) { "TTL must be between 1 and 255." }
        require(dnsResponse.size in DnsTcpForwarder.DNS_HEADER_BYTES..DnsTcpForwarder.MAX_DNS_MESSAGE_BYTES) {
            "DNS response length is outside the supported range."
        }
        val validDnsResponse = try {
            DnsTcpForwarder.validateResponse(query.dnsPayload, dnsResponse)
            true
        } catch (_: Exception) {
            false
        }
        require(validDnsResponse) { "DNS response does not match the original query." }

        val ipHeaderLength = IPV4_MIN_HEADER_BYTES + query.ipOptions.size
        require(ipHeaderLength <= IPV4_MAX_HEADER_BYTES && ipHeaderLength % 4 == 0) {
            "Invalid saved IPv4 options."
        }
        val udpLength = UDP_HEADER_BYTES + dnsResponse.size
        val totalLength = ipHeaderLength + udpLength
        require(totalLength <= MAX_IPV4_PACKET_BYTES) { "DNS response exceeds the IPv4 packet limit." }

        val packet = ByteArray(totalLength)
        packet[0] = ((IPV4_VERSION shl 4) or (ipHeaderLength / 4)).toByte()
        packet[1] = query.dscpEcn.toByte()
        putU16(packet, 2, totalLength)
        putU16(packet, 4, query.identification)
        putU16(packet, 6, if (query.dontFragment) DONT_FRAGMENT_FLAG else 0)
        packet[8] = ttl.toByte()
        packet[9] = UDP_PROTOCOL.toByte()
        query.destinationAddress.copyInto(packet, 12)
        query.sourceAddress.copyInto(packet, 16)
        query.ipOptions.copyInto(packet, IPV4_MIN_HEADER_BYTES)

        val udpOffset = ipHeaderLength
        putU16(packet, udpOffset, query.destinationPort)
        putU16(packet, udpOffset + 2, query.sourcePort)
        putU16(packet, udpOffset + 4, udpLength)
        dnsResponse.copyInto(packet, udpOffset + UDP_HEADER_BYTES)

        var udpSum = udpChecksum(
            query.destinationAddress,
            query.sourceAddress,
            packet,
            udpOffset,
            udpLength
        )
        if (udpSum == 0) udpSum = 0xffff
        putU16(packet, udpOffset + 6, udpSum)
        putU16(packet, 10, checksum(packet, 0, ipHeaderLength))
        return packet
    }

    private fun verifyUdpChecksum(
        source: ByteArray,
        destination: ByteArray,
        packet: ByteArray,
        offset: Int,
        length: Int
    ): Boolean = udpChecksum(source, destination, packet, offset, length) == 0

    private fun udpChecksum(
        source: ByteArray,
        destination: ByteArray,
        packet: ByteArray,
        offset: Int,
        length: Int
    ): Int {
        var sum = 0L
        for (i in 0 until 4 step 2) {
            sum += ((source[i].toInt() and 0xff) shl 8) or (source[i + 1].toInt() and 0xff)
            sum += ((destination[i].toInt() and 0xff) shl 8) or (destination[i + 1].toInt() and 0xff)
        }
        sum += UDP_PROTOCOL
        sum += length
        sum = addWords(sum, packet, offset, length)
        return finishChecksum(sum)
    }

    private fun checksum(bytes: ByteArray, offset: Int, length: Int): Int =
        finishChecksum(addWords(0L, bytes, offset, length))

    private fun addWords(initial: Long, bytes: ByteArray, offset: Int, length: Int): Long {
        var sum = initial
        var index = 0
        while (index + 1 < length) {
            sum += ((bytes[offset + index].toInt() and 0xff) shl 8) or
                (bytes[offset + index + 1].toInt() and 0xff)
            index += 2
        }
        if (index < length) sum += (bytes[offset + index].toInt() and 0xff) shl 8
        return sum
    }

    private fun finishChecksum(value: Long): Int {
        var sum = value
        while ((sum ushr 16) != 0L) sum = (sum and 0xffff) + (sum ushr 16)
        return sum.toInt().inv() and 0xffff
    }

    private fun u8(bytes: ByteArray, offset: Int): Int = bytes[offset].toInt() and 0xff

    private fun u16(bytes: ByteArray, offset: Int): Int =
        (u8(bytes, offset) shl 8) or u8(bytes, offset + 1)

    private fun putU16(bytes: ByteArray, offset: Int, value: Int) {
        bytes[offset] = (value ushr 8).toByte()
        bytes[offset + 1] = value.toByte()
    }

    const val DNS_PORT = 53
    const val DEFAULT_TTL = 64
    private const val IPV4_VERSION = 4
    private const val IPV4_MIN_HEADER_BYTES = 20
    private const val IPV4_MAX_HEADER_BYTES = 60
    private const val UDP_HEADER_BYTES = 8
    private const val MIN_PACKET_BYTES = IPV4_MIN_HEADER_BYTES + UDP_HEADER_BYTES + DnsTcpForwarder.DNS_HEADER_BYTES
    private const val UDP_PROTOCOL = 17
    private const val RESERVED_FLAG = 0x8000
    private const val DONT_FRAGMENT_FLAG = 0x4000
    private const val FRAGMENT_MASK = 0x3fff
    private const val MAX_IPV4_PACKET_BYTES = 65_535
}

package com.aboezzat.vpn.vpn

import java.io.Closeable
import java.util.concurrent.atomic.AtomicBoolean
import java.util.concurrent.atomic.AtomicLong

/**
 * TUN-independent packet loop. A real TUN adapter can implement PacketSource
 * and PacketSink around the file descriptor after this loop is validated.
 *
 * Per-packet DNS/resolver failures are counted and skipped; a source or sink
 * I/O failure exits the loop. This class does not create a VPN interface.
 */
interface DnsPacketSource : Closeable {
    /** Returns the next complete IP packet, or null when the source reaches EOF. */
    fun readPacket(): ByteArray?
}

fun interface DnsPacketSink {
    /** Writes one complete response IP packet back to the packet interface. */
    fun writePacket(packet: ByteArray)
}

class Ipv4UdpDnsPacketLoop(
    private val source: DnsPacketSource,
    private val sink: DnsPacketSink,
    private val processor: Ipv4UdpDnsPacketProcessor,
    private val onPacketError: (Exception) -> Unit = {}
) {
    private val started = AtomicBoolean(false)
    private val stopped = AtomicBoolean(false)
    private val sourceClosed = AtomicBoolean(false)
    private val processedCount = AtomicLong(0)
    private val failedCount = AtomicLong(0)

    val processedPackets: Long get() = processedCount.get()
    val failedPackets: Long get() = failedCount.get()

    /**
     * Runs on the caller's worker thread until EOF, stop(), or an I/O failure.
     * Call stop() from another thread to close a blocking source read.
     */
    fun run() {
        check(started.compareAndSet(false, true)) {
            "Packet loop can only be run once."
        }
        try {
            while (!stopped.get()) {
                val packet = source.readPacket() ?: break
                val response = try {
                    processor.process(packet)
                } catch (error: Exception) {
                    failedCount.incrementAndGet()
                    try {
                        onPacketError(error)
                    } catch (_: Exception) {
                        // Diagnostics must not terminate packet processing.
                    }
                    continue
                }
                sink.writePacket(response)
                processedCount.incrementAndGet()
            }
        } finally {
            closeSource()
        }
    }

    fun stop() {
        stopped.set(true)
        closeSource()
    }

    private fun closeSource() {
        if (sourceClosed.compareAndSet(false, true)) {
            source.close()
        }
    }
}

import java.util.concurrent.ExecutorService
import java.util.concurrent.RejectedExecutionException

/**
 * Owns the native run/stop ordering. The TUN descriptor must stay open for
 * the complete blocking native run call and is closed only after it returns.
 */
internal class NativeTunnelRunner(
    private val executor: ExecutorService,
    private val runEngine: (String, Int, Boolean) -> Int,
    private val stopEngine: () -> Int,
    private val onFinished: (Int) -> Unit
) {
    private val lock = Any()
    private var active = false

    fun isActive(): Boolean = synchronized(lock) { active }

    fun start(
        cliArgs: String,
        mtu: Int,
        packetInformation: Boolean = false,
        closeTunDescriptor: () -> Unit
    ): Boolean = synchronized(lock) {
        if (active) return@synchronized false
        active = true
        try {
            executor.execute {
                val result = try {
                    runEngine(cliArgs, mtu, packetInformation)
                } catch (_: Throwable) {
                    NATIVE_RUN_EXCEPTION
                } finally {
                    // Do not close the Android PFD until native run has
                    // completely returned and released its raw-fd use.
                }
                try {
                    closeTunDescriptor()
                } finally {
                    synchronized(lock) { active = false }
                    onFinished(result)
                }
            }
            true
        } catch (_: RejectedExecutionException) {
            active = false
            false
        }
    }

    fun stop(): Boolean = synchronized(lock) {
        if (!active) return@synchronized false
        try {
            stopEngine() >= 0
        } catch (_: Throwable) {
            false
        }
    }

    companion object {
        const val NATIVE_RUN_EXCEPTION = -100
    }
}

package com.aboezzat.vpn.vpn

/**
 * Bounded bookkeeping for observed TCP flow lifecycles.
 *
 * This is deliberately NOT a TCP implementation: it does not generate ACKs,
 * retransmit segments, negotiate MSS/window scaling, reorder data, or create
 * upstream sockets. It only tracks packet observations for diagnostics and
 * future integration with a real TCP/IP stack.
 */
class TcpFlowLifecycleTracker(
    private val maxFlows: Int = DEFAULT_MAX_FLOWS,
    private val closedRetentionMillis: Long = DEFAULT_CLOSED_RETENTION_MILLIS
) {
    enum class Direction { OUTBOUND, INBOUND }

    enum class State {
        SYN_SENT,
        SYN_ACK_RECEIVED,
        ESTABLISHED,
        LOCAL_HALF_CLOSED,
        REMOTE_HALF_CLOSED,
        CLOSING,
        CLOSED
    }

    data class FlowKey(
        val localAddress: String,
        val localPort: Int,
        val remoteAddress: String,
        val remotePort: Int
    )

    data class Snapshot(
        val key: FlowKey,
        val state: State,
        val createdAtMillis: Long,
        val lastSeenAtMillis: Long,
        val localFinSeen: Boolean,
        val remoteFinSeen: Boolean,
        val lastSequenceNumber: Long,
        val lastAcknowledgementNumber: Long,
        val packetsSeen: Long,
        val payloadBytesSeen: Long
    )

    sealed interface ObserveResult {
        data class Updated(val snapshot: Snapshot) : ObserveResult
        data object CapacityReached : ObserveResult
        data object InvalidInitialPacket : ObserveResult
    }

    private data class MutableFlow(
        val key: FlowKey,
        val createdAtMillis: Long,
        var lastSeenAtMillis: Long,
        var state: State,
        var localFinSeen: Boolean = false,
        var remoteFinSeen: Boolean = false,
        var lastSequenceNumber: Long = 0,
        var lastAcknowledgementNumber: Long = 0,
        var packetsSeen: Long = 0,
        var payloadBytesSeen: Long = 0
    ) {
        fun snapshot() = Snapshot(
            key, state, createdAtMillis, lastSeenAtMillis,
            localFinSeen, remoteFinSeen, lastSequenceNumber,
            lastAcknowledgementNumber, packetsSeen, payloadBytesSeen
        )
    }

    private val flows = LinkedHashMap<FlowKey, MutableFlow>()

    init {
        require(maxFlows > 0) { "maxFlows must be positive." }
        require(closedRetentionMillis >= 0) { "closedRetentionMillis cannot be negative." }
    }

    @Synchronized
    fun observe(packet: Ipv4TcpPacketParser.Packet, direction: Direction, nowMillis: Long): ObserveResult {
        require(nowMillis >= 0) { "nowMillis cannot be negative." }
        prune(nowMillis)

        val key = when (direction) {
            Direction.OUTBOUND -> FlowKey(
                packet.tuple.local.address.hostAddress!!,
                packet.tuple.local.port,
                packet.tuple.remote.address.hostAddress!!,
                packet.tuple.remote.port
            )
            Direction.INBOUND -> FlowKey(
                packet.tuple.remote.address.hostAddress!!,
                packet.tuple.remote.port,
                packet.tuple.local.address.hostAddress!!,
                packet.tuple.local.port
            )
        }

        val isNewOutboundSyn = direction == Direction.OUTBOUND &&
            packet.isSyn && !packet.isAck && !packet.isRst
        var flow = flows[key]

        // A terminal diagnostic record must not absorb a new TCP connection
        // that reuses the same four-tuple.
        if (flow != null && isNewOutboundSyn && flow.state.isTerminal()) {
            flows.remove(key)
            flow = null
        }

        if (flow == null) {
            if (!isNewOutboundSyn) return ObserveResult.InvalidInitialPacket
            if (flows.size >= maxFlows) return ObserveResult.CapacityReached
            flow = MutableFlow(
                key = key,
                createdAtMillis = nowMillis,
                lastSeenAtMillis = nowMillis,
                state = State.SYN_SENT
            )
            flows[key] = flow
        }

        // Keep timestamps monotonic if callers deliver delayed observations.
        flow.lastSeenAtMillis = maxOf(flow.lastSeenAtMillis, nowMillis)
        flow.lastSequenceNumber = packet.sequenceNumber
        flow.lastAcknowledgementNumber = packet.acknowledgementNumber
        flow.packetsSeen = saturatingAdd(flow.packetsSeen, 1)
        flow.payloadBytesSeen = saturatingAdd(flow.payloadBytesSeen, packet.payload.size.toLong())

        if (packet.isRst) {
            flow.state = State.CLOSED
        } else {
            if (packet.isFin) {
                if (direction == Direction.OUTBOUND) flow.localFinSeen = true
                else flow.remoteFinSeen = true
            }

            if (flow.localFinSeen && flow.remoteFinSeen) {
                flow.state = State.CLOSING
            } else if (flow.localFinSeen) {
                flow.state = State.LOCAL_HALF_CLOSED
            } else if (flow.remoteFinSeen) {
                flow.state = State.REMOTE_HALF_CLOSED
            } else {
                flow.state = when {
                    flow.state == State.SYN_SENT && direction == Direction.INBOUND &&
                        packet.isSyn && packet.isAck -> State.SYN_ACK_RECEIVED
                    flow.state == State.SYN_ACK_RECEIVED && direction == Direction.OUTBOUND &&
                        packet.isAck && !packet.isSyn -> State.ESTABLISHED
                    flow.state == State.SYN_SENT && direction == Direction.OUTBOUND &&
                        packet.isSyn -> State.SYN_SENT
                    flow.state == State.ESTABLISHED -> State.ESTABLISHED
                    else -> flow.state
                }
            }
        }

        return ObserveResult.Updated(flow.snapshot())
    }

    @Synchronized
    fun snapshot(): List<Snapshot> = flows.values.map { it.snapshot() }

    @Synchronized
    fun remove(key: FlowKey): Boolean = flows.remove(key) != null

    @Synchronized
    fun prune(nowMillis: Long): Int {
        require(nowMillis >= 0) { "nowMillis cannot be negative." }
        val before = flows.size
        val iterator = flows.entries.iterator()
        while (iterator.hasNext()) {
            val flow = iterator.next().value
            if (flow.state.isTerminal() &&
                nowMillis >= flow.lastSeenAtMillis &&
                nowMillis - flow.lastSeenAtMillis >= closedRetentionMillis
            ) {
                iterator.remove()
            }
        }
        return before - flows.size
    }

    private fun State.isTerminal(): Boolean = this == State.CLOSING || this == State.CLOSED

    private fun saturatingAdd(current: Long, increment: Long): Long =
        if (Long.MAX_VALUE - current < increment) Long.MAX_VALUE else current + increment

    companion object {
        const val DEFAULT_MAX_FLOWS = 512
        const val DEFAULT_CLOSED_RETENTION_MILLIS = 30_000L
    }
}

package com.aboezzat.vpn.vpn

import com.aboezzat.vpn.proxy.ProxyProfile
import com.aboezzat.vpn.proxy.ProxyProtocol

/**
 * Builds the exact CLI consumed by the vendored tun2proxy 0.8.7 parser.
 * Credentials are percent-encoded before entering the proxy URL and every
 * argument is shell-quoted for shlex::split on the Rust side.
 */
internal object Tun2ProxyCommandBuilder {
    fun build(profile: ProxyProfile, tunFd: Int, mtu: Int = 1500): String? {
        if (profile.validate() != null || tunFd < 0 || mtu !in 576..65535) return null

        val scheme = when (profile.protocol) {
            ProxyProtocol.HTTP_CONNECT -> "http"
            ProxyProtocol.SOCKS5 -> "socks5"
        }
        val host = profile.host
            .removePrefix("[")
            .removeSuffix("]")
            .let { value -> if (value.contains(':')) "[$value]" else value }

        val auth = if (profile.username != null && profile.password != null) {
            "${percentEncode(profile.username)}:${percentEncode(profile.password)}@"
        } else {
            ""
        }
        val proxyUrl = "$scheme://$auth$host:${profile.port}"

        val dnsMode = if (profile.remoteDns) "virtual" else "over-tcp"
        val arguments = mutableListOf(
            "tun2proxy-bin",
            "--tun-fd", tunFd.toString(),
            "--close-fd-on-drop", "false",
            "--proxy", proxyUrl,
            "--dns", dnsMode,
            "--ipv6-enabled",
            "--mtu", mtu.toString(),
            "--verbosity", "info"
        )

        // The native engine intentionally drops non-DNS UDP unless opted in.
        // Android Proxy Server's SOCKS5 UDP ASSOCIATE path is tested separately,
        // so enable real UDP forwarding for SOCKS5 profiles (QUIC, messaging,
        // speed tests, etc.). HTTP CONNECT has no UDP transport and stays TCP-only.
        if (profile.protocol == ProxyProtocol.SOCKS5) {
            arguments += "--udp-enabled"
        }

        return arguments.joinToString(" ") { shellQuote(it) }
    }

    private fun percentEncode(value: String): String {
        val hex = "0123456789ABCDEF"
        val out = StringBuilder()
        value.toByteArray(Charsets.UTF_8).forEach { byte ->
            val valueByte = byte.toInt() and 0xff
            val safe = valueByte in 0x41..0x5a ||
                valueByte in 0x61..0x7a ||
                valueByte in 0x30..0x39 ||
                valueByte == 0x2d || valueByte == 0x2e ||
                valueByte == 0x5f || valueByte == 0x7e
            if (safe) {
                out.append(valueByte.toChar())
            } else {
                out.append('%')
                out.append(hex[valueByte ushr 4])
                out.append(hex[valueByte and 0x0f])
            }
        }
        return out.toString()
    }

    private fun shellQuote(value: String): String =
        "'" + value.replace("'", "'\\''") + "'"
}

            result[index] = value.toByte()

            matched = if (char == ending[matched]) matched + 1
            else if (char == '\r') 1 else 0
        }
    }
